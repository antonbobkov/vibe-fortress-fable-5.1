let seed = 987654321;
const rnd = (): number => { seed ^= seed << 13; seed >>>= 0; seed ^= seed >>> 17; seed ^= seed << 5; seed >>>= 0; return seed / 0x100000000; };
const pick = <T>(xs: readonly T[]): T => xs[Math.floor(rnd() * xs.length)] as T;

function poolFor(field: string, p: Pools): unknown[] {
  if (p.byField[field]) return p.byField[field] as unknown[];
  if (/Id$/.test(field)) return p.ids;
  return p.garbage;
}

function show(v: unknown): string {
  const s = JSON.stringify(v, (_k, x) => (typeof x === 'number' && !Number.isFinite(x) ? `#${String(x)}` : x === undefined ? '#undefined' : typeof x === 'function' ? '#fn' : Array.isArray(x) && x.length > 12 ? `#array(${x.length})` : typeof x === 'string' && x.length > 40 ? `#string(${x.length})` : x));
  return s === undefined ? '#undefined' : s.length > 300 ? `${s.slice(0, 300)}…` : s;
}

describe('T-115 probe: command fuzz', () => {
  it('every command with boundary and malformed arguments', () => {
    mkdirSync(OUT, { recursive: true });
    const base = buildWorld();
    const baseGame = base.game;
    const caravan = base.events('CaravanAtDepot').length > 0;
    const lines: string[] = [`base world at tick ${baseGame.tick}; caravan at depot: ${caravan}; trials per command: field sweep + ${N} random`];
    const known = new Set<string>(REJECT_REASONS);
    const baseline = new Probe('baseline (before any fuzzing)'); baseline.check(baseGame);
    const loadedBase = roundTrip(baseGame).game; const baseline2 = new Probe('baseline after one save/load'); baseline2.check(loadedBase);
    lines.push(baseline.report(), baseline2.report(), `hash equal after load: ${loadedBase.hash() === baseGame.hash()}`);
    const summary: string[] = [];
    for (const type of COMMAND_TYPES) {
      if (ONLY.length > 0 && !ONLY.includes(type)) continue;
      let game = roundTrip(baseGame).game;
      let p = pools(game);
      let probe = new Probe(type, 3);
      const owner = COMMAND_OWNERS[type];
      const fields = FIELDS[owner] ?? [];
      const own = OWN[type] ?? fields;
      const results: Record<string, number> = {};
      const throws = new Map<string, { n: number; cmd: string }>();
      const accepted = new Map<string, string>();
      const unknownReasons = new Set<string>();
      const invHits = new Map<string, string>();
      let saveDiverged: string[] = [];
      let acceptedSince: string[] = [];
      let trials = 0;
      const pre = new Set<string>();
      const prime = (): void => { probe.check(game); for (const k of probe.findings.keys()) pre.add(k); };
      prime();
      const refork = (): void => { game = roundTrip(baseGame).game; p = pools(game); probe = new Probe(type, 3); acceptedSince = []; prime(); };
      const trial = (cmd: Record<string, unknown>): void => {
        trials += 1;
        const text = show(cmd);
        let res: { ok: boolean; reason?: string } | undefined;
        try {
          res = game.world.applyCommand(cmd as unknown as Command) as { ok: boolean; reason?: string };
        } catch (err) {
          const msg = `apply: ${(err as Error).message?.slice(0, 120)} @ ${String((err as Error).stack).split('\n')[1]?.trim().slice(0, 140)}`;
          const t = throws.get(msg) ?? { n: 0, cmd: text }; t.n += 1; throws.set(msg, t);
          results['THROW'] = (results['THROW'] ?? 0) + 1;
          refork();
          return;
        }
        const key = res.ok ? 'ok' : `reject:${String(res.reason)}`;
        results[key] = (results[key] ?? 0) + 1;
        if (!res.ok && !known.has(String(res.reason))) unknownReasons.add(String(res.reason));
        if (res.ok) {
          const shape = Object.keys(cmd).filter((k) => k !== 'type').map((k) => `${k}:${show(cmd[k]).slice(0, 60)}`).join(' ');
          if (accepted.size < 400) accepted.set(shape, text);
          acceptedSince.push(text);
          try { game.step(2); } catch (err) {
            const msg = `step after accept: ${(err as Error).message?.slice(0, 120)} @ ${String((err as Error).stack).split('\n')[1]?.trim().slice(0, 140)}`;
            const t = throws.get(msg) ?? { n: 0, cmd: text }; t.n += 1; throws.set(msg, t);
            results['THROW-IN-STEP'] = (results['THROW-IN-STEP'] ?? 0) + 1;
            refork();
            return;
          }
          const before = probe.findings.size;
          try { probe.check(game); } catch (err) { invHits.set(`probe threw: ${(err as Error).message}`, text); refork(); return; }
          if (probe.findings.size > before) for (const [fk, f] of probe.findings) if (!pre.has(fk) && !invHits.has(`${f.inv} [${f.cls}]`)) invHits.set(`${f.inv} [${f.cls}]`, `${f.samples[0]} <= after ${text}`);
          if (acceptedSince.length >= 15) {
            try {
              const copy = roundTrip(game).game;
              if (copy.hash() !== game.hash()) { saveDiverged.push(`hash differs after a save/load; accepted since the last check: ${acceptedSince.join(' ;; ')}`); refork(); }
            } catch (err) { saveDiverged.push(`save/load threw ${(err as Error).message}; accepted: ${acceptedSince.join(' ;; ')}`); refork(); }
            acceptedSince = [];
          }
        }
      };
      const baseCmd = (): Record<string, unknown> => {
        const cmd: Record<string, unknown> = { type };
        for (const f of own) cmd[f] = (poolFor(f, p)[0]);
        return cmd;
      };
      // 0. no fields at all, and the documented fields each missing
      trial({ type });
      // 1. single-field sweep over a plausible base, three bases
      for (let b = 0; b < 6; b += 1) {
        const start = baseCmd();
        if (b > 0) for (const f of own) start[f] = pick(poolFor(f, p).slice(0, p.goodCount[f] ?? 6));
        for (const f of own) for (const v of poolFor(f, p)) trial({ ...start, [f]: v });
      }
      // 2. random
      for (let k = 0; k < N; k += 1) {
        const cmd: Record<string, unknown> = { type };
        for (const f of own) if (rnd() < 0.92) cmd[f] = rnd() < 0.5 ? pick(poolFor(f, p).slice(0, p.goodCount[f] ?? 6)) : rnd() < 0.8 ? pick(poolFor(f, p)) : pick(p.garbage);
        if (rnd() < 0.15) { const extra = pick(fields); if (extra !== undefined) cmd[extra] = pick(poolFor(extra, p)); }
        trial(cmd);
      }
      if (acceptedSince.length > 0) {
        try { const copy = roundTrip(game).game; if (copy.hash() !== game.hash()) saveDiverged.push(`hash differs after a save/load; accepted since the last check: ${acceptedSince.join(' ;; ')}`); }
        catch (err) { saveDiverged.push(`save/load threw ${(err as Error).message}; accepted: ${acceptedSince.join(' ;; ')}`); }
      }
      const flag = throws.size > 0 || unknownReasons.size > 0 || invHits.size > 0 || saveDiverged.length > 0;
      summary.push(`${flag ? '!!' : '  '} ${type} (${owner}): ${trials} trials ${JSON.stringify(results)}${unknownReasons.size ? ` UNKNOWN REASONS ${[...unknownReasons].join(',')}` : ''}${throws.size ? ` THROWS ${throws.size}` : ''}${invHits.size ? ` INVARIANTS ${invHits.size}` : ''}${saveDiverged.length ? ` SAVE ${saveDiverged.length}` : ''}`);
      lines.push(`\n===== ${type} (${owner}) ${trials} trials ${JSON.stringify(results)}`);
      for (const [msg, t] of throws) lines.push(`  THROW x${t.n}: ${msg}\n      first: ${t.cmd}`);
      for (const r of unknownReasons) lines.push(`  UNKNOWN REASON: ${r}`);
      for (const [k, v] of invHits) lines.push(`  INVARIANT ${k}: ${v}`);
      for (const s of saveDiverged.slice(0, 3)) lines.push(`  SAVE: ${s.slice(0, 3000)}`);
      lines.push(`  accepted shapes (${accepted.size}):`);
      let shown = 0;
      for (const text of accepted.values()) { if (shown >= 60) break; lines.push(`    ${text}`); shown += 1; }
    }
    writeFileSync(`${OUT}/fuzz.txt`, [...summary, ...lines].join('\n'));
    console.log(summary.join('\n'));
  }, 3_600_000);
});
