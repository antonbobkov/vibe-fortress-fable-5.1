/**
 * T-427 probe (not committed; filed as reports/data/2026-10-23-t427-presence.probe.ts.txt): the annual
 * instrument's twelve years, sampling every look (20 ticks) the wild animals by species, those present
 * (any state but `leave`) and those leaving. Prints the mean per year for each seed and the cohort.
 */
import { describe, it } from 'vitest';
import { writeFileSync } from 'node:fs';
import { C } from '@core/components';
import { playWildlifeYear, WILDLIFE_CASES } from './wildlife_year_fixture';

const OUT = process.env['T427_OUT'] ?? '.vite/t427-presence.json';

describe('T-427 presence', () => {
  it('samples the wild population', () => {
    const rows: Record<string, unknown>[] = [];
    for (const { seed, biome } of WILDLIFE_CASES) {
      const present: Record<string, number> = {}; const leaving: Record<string, number> = {}; const hunting: Record<string, number> = {};
      let samples = 0;
      const since = new Map<number, { sp: string; from: number }>(); const held: Record<string, number[]> = {};
      let lastTick = 0;
      const row = playWildlifeYear(seed, biome, {
        observe: (game) => {
          samples += 1;
          const e = game.world.entities; const table = game.registry.table('creatures');
          lastTick = game.tick;
          const now = new Set<number>();
          for (const id of e.with(C.Wild)) {
            const sp = table.idOf(e.get(id, C.Creature)!.species);
            const bucket = e.get(id, C.Wild)?.state === 'leave' ? leaving : present;
            if (bucket === leaving) { now.add(id); if (!since.has(id)) since.set(id, { sp, from: game.tick }); }
            bucket[sp] = (bucket[sp] ?? 0) + 1;
            const st = e.get(id, C.Wild)?.state;
            if (st === 'hunt' || st === 'attack') hunting[sp] = (hunting[sp] ?? 0) + 1;
          }
          for (const [id, v] of [...since]) if (!now.has(id)) { (held[v.sp] ??= []).push(game.tick - v.from); since.delete(id); }
        },
      });
      for (const v of since.values()) (held[v.sp] ??= []).push(lastTick - v.from);
      const leaveDays = Object.fromEntries(Object.entries(held).sort().map(([k, v]) => [k, { n: v.length, meanDays: +(v.reduce((a, b) => a + b, 0) / v.length / 1200).toFixed(2), over1: v.filter((d) => d > 1200).length }]));
      const mean = (o: Record<string, number>): Record<string, number> => Object.fromEntries(Object.entries(o).sort().map(([k, v]) => [k, +(v / samples).toFixed(2)]));
      const r = { seed, biome, living: row.living, spawns: row.spawns, present: mean(present), leaving: mean(leaving), hunting: mean(hunting), deaths: row.deaths.map((d) => `${d.day}:${d.killerSpecies}`), leaveDays };
      console.log(`[t427-presence] ${JSON.stringify(r)}`);
      rows.push(r);
    }
    writeFileSync(OUT, JSON.stringify(rows, null, 1));
  }, 7_200_000);
});
