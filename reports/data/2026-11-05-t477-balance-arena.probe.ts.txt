/**
 * T-477 balance probe (not committed as a test; a copy is filed under reports/data as
 * `…-t477-balance-arena.probe.ts.txt`; run from test/perf). The combat-pace squad arenas of T-366,
 * T-473 and T-483 on the flat test map, through the real tick loop, on the same seeds
 * (`t366-squad-<arena>-<i>`): five Novice soldiers in full kit (iron helm, mail shirt, greaves, axe,
 * shield) at a station, and in place of a sixth:
 *  - `ward`: a Proficient runecaster (runecraft 10, rune 50, Novice dodge, a tunic, no metal) holding
 *    a common iron staff of warding (a ward of 7 for 720 ticks);
 *  - `daze`: the same caster with a common iron staff of stillness (Daze, 180 ticks, 3 rune);
 *  - `five`: no one;
 *  - `six`: one more axedwarf in full kit;
 *  - `marks`: a Proficient marksdwarf (crossbow 10, a tunic, an oak crossbow, 25 iron bolts) in a
 *    ranged squad of its own at the same station;
 *  - `seven`: two more axedwarves (a squad of six and a second of one at the same station);
 *  - `sixmarks`: one more axedwarf and the marksdwarf.
 * open: a peril-2 raid of ten in the open; ambush: a peril-2 ambush of four. Three civilians stand
 * eight tiles behind. Each trial runs until one side is down or the party leaves (at most 4,800
 * ticks), then a day with no medic. Writes T477_OUT (JSON lines). Observation only.
 */
import { describe, it } from 'vitest';
import { appendFileSync, writeFileSync } from 'node:fs';
import { FLAT_TEST_SURFACE_Z } from '@app/headless';
import { C, type Equipment } from '@core/components';
import type { EntityId } from '@core/types';
import type { CombatReader, SquadWithId } from '@systems/combat/reader';
import { arriveParty, spawnHostileMember } from '@systems/threats/parties';
import { TestWorld } from '../scenarios/test-world';

const OUT = process.env['T477_OUT'] ?? '.vite/t477-balance.jsonl';
const N = Number(process.env['T477_N'] ?? 24);
const ONLY = process.env['T477_ONLY'] ?? '';
const Z = FLAT_TEST_SURFACE_Z;

interface Kit { weapon: [string, string]; shield?: [string, string]; body?: [string, string]; head?: [string, string]; legs?: [string, string] }
const FULL_KIT: Kit = { weapon: ['axe', 'iron'], shield: ['shield', 'iron'], body: ['mail_shirt', 'iron'], head: ['helm', 'iron'], legs: ['greaves', 'iron'] };
const NOVICE = { axe: 2, shield: 2, dodge: 2, armor_use: 2, hammer: 2 };

function equip(w: TestWorld, id: EntityId, at: number, kit: Kit): void {
  const items = w.game.registry.table('items');
  const mats = w.game.registry.table('materials');
  const make = (k?: [string, string]): EntityId => (k ? w.spawnItem(items.index(k[0]), mats.index(k[1]), at, 1, 1) : 0);
  const old = w.game.world.entities.get(id, C.Equipment);
  const eq: Equipment = {
    weapon: make(kit.weapon), shield: make(kit.shield),
    armor: { head: make(kit.head), body: make(kit.body), arms: 0, legs: make(kit.legs), feet: 0 },
    clothing: old?.clothing ?? { body: 0, legs: 0, feet: 0, over: 0 }, quiver: 0,
  };
  w.game.world.entities.add(id, C.Equipment, eq);
}

function squad(w: TestWorld, name: string, members: EntityId[], station: number, ranged = false): void {
  w.game.world.applyCommand({ type: 'CreateSquad', name });
  const all = w.readers.of<CombatReader>('16').squads();
  const s = all[all.length - 1] as SquadWithId;
  w.game.world.applyCommand({ type: 'SetSquadMembers', squadId: s.id, members });
  if (ranged) w.game.world.applyCommand({ type: 'SetSquadEquipment', squadId: s.id, equipment: { weaponClass: 'any', armor: 'any', ranged: true } });
  w.game.world.applyCommand({ type: 'SetSquadStation', squadId: s.id, tile: station });
  w.game.world.applyCommand({ type: 'SetSquadMode', squadId: s.id, mode: 'active' });
}

function standing(w: TestWorld, id: EntityId): boolean {
  const ent = w.game.world.entities;
  if (!ent.alive(id) || !ent.get(id, C.Position)) return false;
  return ent.get(id, C.Body)?.conscious !== false;
}

/** Blood lost: a dead dwarf has lost all of it. */
function bloodLost(w: TestWorld, id: EntityId, max: number): number {
  const b = w.game.world.entities.get(id, C.Body);
  return b ? b.maxBlood - b.blood : max;
}

function trial(kind: string, i: number): Record<string, unknown> {
  const [base, variant] = kind.split('-') as ['open' | 'ambush', string];
  const w = new TestWorld({ seed: `t366-squad-${base}-${i}`, world: 'flat-test', embark: 'none', surfaceWildlife: 'none' });
  const reg = w.game.registry;
  const ent = w.game.world.entities;
  const sx = 46, sy = 60;
  const axes = variant === 'six' || variant === 'sixmarks' ? 6 : variant === 'seven' ? 7 : 5;
  const ids: EntityId[] = [];
  for (let k = 0; k < axes; k += 1) ids.push(w.spawnDwarf({ at: w.index(sx + k, sy, Z), skills: NOVICE }));
  const spell = variant === 'ward' ? 'quartz' : variant === 'daze' ? 'diamond' : '';
  const mage = spell === '' ? 0 : w.spawnDwarf({ at: w.index(sx + axes, sy, Z), skills: { runecraft: 10, dodge: 2, hammer: 2 } });
  if (mage !== 0) ent.add(mage, C.Rune, { current: 50_000 });
  const marks = variant === 'marks' || variant === 'sixmarks' ? w.spawnDwarf({ at: w.index(sx + axes, sy + 1, Z), skills: { crossbow: 10, dodge: 2, hammer: 2 } }) : 0;
  const station = w.index(sx, sy, Z);
  const melee = mage === 0 ? ids : [...ids, mage];
  squad(w, 'Guard', melee.slice(0, 6), station);
  if (melee.length > 6) squad(w, 'Second', melee.slice(6), station);
  if (marks !== 0) squad(w, 'Bolts', [marks], station, true);
  w.step(1);
  ids.forEach((id, k) => equip(w, id, w.index(sx + k, sy, Z), FULL_KIT));
  if (mage !== 0) {
    const item = w.spawnItem('staff', 'iron', w.index(sx + axes, sy, Z), 1, 1);
    ent.add(item, C.Item, { ...ent.get(item, C.Item)!, stone: reg.table('materials').index(spell) });
    ent.add(mage, C.Equipment, { ...ent.get(mage, C.Equipment)!, weapon: item });
  }
  if (marks !== 0) {
    const items = w.game.world.services.items!;
    const bow = w.spawnItem('crossbow', 'oak', w.index(sx + axes, sy + 1, Z), 1, 1);
    const bolts = w.spawnItem('bolt', 'iron', w.index(sx + axes, sy + 1, Z), 25, 1);
    for (const it of [bow, bolts]) { items.attachToCarrier(it, marks); items.setOwner(it, marks); }
    ent.add(marks, C.Equipment, { ...ent.get(marks, C.Equipment)!, weapon: bow, quiver: bolts });
  }
  const maxBlood = ent.get(ids[0]!, C.Body)!.maxBlood;
  const civ: EntityId[] = [];
  for (let k = 0; k < 3; k += 1) civ.push(w.spawnDwarf({ at: w.index(sx + k, sy + 8, Z) }));
  const rng = w.game.world.rngFor('t366');
  const ambush = base === 'ambush';
  const { partyId, pendingSpawns } = arriveParty(w.game.world as never, reg, rng, ambush ? 'ambush' : 'raid', 'gloam', ambush ? 4 : 10, 'n', w.tick, 15, 2, false);
  for (const s of pendingSpawns) spawnHostileMember(w.game.world as never, reg, rng, s.partyId, s.species, s.role, s.tile, s.gearTier, s.isCaptain);
  const members = [...ent.get(partyId, C.Party)!.members];
  const captain = members.find((m) => ent.get(m, C.Hostile)?.role === 'captain') ?? 0;
  const fighters = [...melee, ...(marks === 0 ? [] : [marks])];
  const tally = { wards: 0, warded: 0, dazes: 0, captainDazed: 0, captainStunTicks: 0, raiderHits: 0, raiderDamage: 0, raidersFled: 0, events: {} as Record<string, number> };
  let ended = '';
  let captainFate = 'standing';
  let captainLostTick = -1;
  w.game.world.onEvents((ev) => {
    for (const e of ev) {
      if (e.type === 'SpellCast' && Number(e['casterId']) === mage) {
        if (e['spellId'] === 'ward') tally.wards += 1;
        if (e['spellId'] === 'daze') tally.dazes += 1;
      } else if (e.type === 'Warded') tally.warded += 1;
      else if (e.type === 'CreatureStunned' && Number(e['id']) === captain) { tally.captainDazed += 1; tally.captainStunTicks += Number(e['until']) - w.tick; }
      else if (e.type === 'RaidEnded' && Number(e['partyId']) === partyId) ended = String(e['outcome']);
      else if (e.type === 'HostileFled' && members.includes(Number(e['id']))) tally.raidersFled += 1;
      else if (e.type === 'AttackResolved' && members.includes(Number(e['attackerId'])) && fighters.includes(Number(e['defenderId'])) && e['outcome'] === 'hit') {
        tally.raiderHits += 1; tally.raiderDamage += Number(e['amount']);
      }
      if ((e.type === 'CreatureDied' || e.type === 'CreatureUnconscious') && Number(e['id']) === captain && captainFate === 'standing') {
        captainFate = e.type === 'CreatureDied' ? 'died' : 'down'; captainLostTick = w.tick;
      }
      // Whatever T-504 announces about a raid's nerve and morale is counted by type.
      if (/Nerve|Morale|Rout|Broke/.test(e.type)) tally.events[e.type] = (tally.events[e.type] ?? 0) + 1;
    }
  });
  let endTick = 0;
  for (let t = 0; t < 4800; t += 1) {
    w.step(1);
    if (ended !== '' || members.every((m) => !standing(w, m)) || fighters.every((d) => !standing(w, d))) { endTick = w.tick; break; }
  }
  const after = (): Record<string, unknown> => ({
    soldiersDead: ids.filter((d) => !ent.alive(d)).length,
    extraDead: [mage, marks].filter((d) => d !== 0 && !ent.alive(d)).length,
    civDead: civ.filter((d) => !ent.alive(d)).length,
    raidersDead: members.filter((m) => !ent.alive(m)).length,
    blood: fighters.reduce((s, d) => s + bloodLost(w, d, maxBlood), 0),
    soldierBlood: ids.slice(0, 5).reduce((s, d) => s + bloodLost(w, d, maxBlood), 0),
  });
  const atEnd = after();
  w.step(1200);
  return {
    kind, i, raiders: members.length, captain: captain !== 0, captainFate, captainLostTick,
    outcome: ended || (fighters.every((d) => !standing(w, d)) ? 'squad_down' : members.every((m) => !standing(w, m)) ? 'raid_down' : 'timeout'),
    endTick, ...atEnd, dayAfter: after(), ...tally,
  };
}

describe('T-477 balance probe', () => {
  it('runs the arenas', () => {
    if (!ONLY) writeFileSync(OUT, '');
    const put = (r: Record<string, unknown>): void => appendFileSync(OUT, JSON.stringify(r) + '\n');
    const kinds = (ONLY || 'open-five,open-ward,open-daze,open-six,open-marks,open-seven,open-sixmarks,ambush-five,ambush-ward,ambush-six,ambush-seven').split(',');
    for (const k of kinds) for (let i = 0; i < N; i += 1) put(trial(k, i));
  }, 7_200_000);
});
