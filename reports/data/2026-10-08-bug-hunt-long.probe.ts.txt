/* T-115 scratch: the three-year fixture with invariant checks, hash trace, and save/load forks. NOT a permanent test.
 * env: T115_SEED, T115_PERIL, T115_DAYS (default 336), T115_OUT (dir), T115_SAVE=1 (fork a loaded copy at chosen ticks),
 *      T115_EVERY (check cadence in ticks, default 50), T115_TAG
 */
import { describe, it } from 'vitest';
import { mkdirSync, writeFileSync } from 'node:fs';
import { TICKS_PER_DAY } from '@core/constants';
import { C } from '@core/components';
import { roundTrip } from '@app/game-save';
import type { HeadlessGame } from '@app/headless';
import { TestWorld } from '../test/scenarios/test-world';
import { stepThreeYears, attachThreeYearsMeasurements } from '../test/scenarios/three_years_measurements';
import { setupThreeYearsFortress, checkFortressWorkOrders, checkFortressRooms, checkFortressCaveFarm, checkFortressDefense, threeYearsWorld, type FortressState } from '../test/scenarios/three_years_fortress';
import { Probe } from './invariants';

const SEED = process.env['T115_SEED'] ?? 'three-years';
const PERIL = Number(process.env['T115_PERIL'] ?? '0');
const DAYS = Number(process.env['T115_DAYS'] ?? '336');
const OUT = process.env['T115_OUT'] ?? 't115-scratch/out';
const SAVE = process.env['T115_SAVE'] === '1';
const EVERY = Number(process.env['T115_EVERY'] ?? '50');
const TAG = process.env['T115_TAG'] ?? 'a';
const FORK_TICKS = 1200;

function script(w: TestWorld, state: FortressState): void {
  checkFortressWorkOrders(w, state);
  checkFortressRooms(w, state);
  checkFortressCaveFarm(w, state);
  checkFortressDefense(w, state);
}

function wrap(game: HeadlessGame): TestWorld {
  const w = Object.create(TestWorld.prototype) as TestWorld;
  Object.defineProperty(w, 'game', { value: game });
  Object.defineProperty(w, 'seen', { value: [] });
  attachThreeYearsMeasurements(w);
  return w;
}


/** Where two games differ, section by section, in the order the world hash reads them. */
function whyDiffer(a: HeadlessGame, b: HeadlessGame): string {
  const out: string[] = [];
  const canon = (x: unknown): unknown => {
    if (ArrayBuffer.isView(x)) return Array.from(x as unknown as ArrayLike<number>);
    if (x instanceof Map) return Array.from(x.entries()).map(canon);
    if (x instanceof Set) return Array.from(x.values()).map(canon);
    if (Array.isArray(x)) return x.map(canon);
    if (x !== null && typeof x === 'object') { const o: Record<string, unknown> = {}; for (const k of Object.keys(x as object).sort()) { const v = (x as Record<string, unknown>)[k]; if (v !== undefined) o[k] = canon(v); } return o; }
    return typeof x === 'number' && !Number.isFinite(x) ? `#${String(x)}` : x;
  };
  const J = (v: unknown): string => JSON.stringify(canon(v)) ?? 'undefined';
  for (const grid of ['kind', 'material', 'fluid', 'designation', 'flags', 'tempOverride'] as const) {
    const ga = a.map[grid] as ArrayLike<number>; const gb = b.map[grid] as ArrayLike<number>;
    let n = 0; const bits: Record<number, number> = {}; let first = -1;
    for (let i = 0; i < ga.length; i += 1) if (ga[i] !== gb[i]) { n += 1; if (first < 0) first = i; const x = (ga[i] as number) ^ (gb[i] as number); for (let k = 0; k < 16; k += 1) if (x & (1 << k)) bits[k] = (bits[k] ?? 0) + 1; }
    if (n > 0) { const p = a.map.unpack(first); out.push(`map.${grid}: ${n} tiles differ, bits ${JSON.stringify(bits)}, first at (${p.x},${p.y},${p.z}) ${ga[first]} vs ${gb[first]}`); }
  }
  const ia = a.world.entities.allIds(); const ib = b.world.entities.allIds();
  if (J(ia) !== J(ib)) out.push(`entity ids differ: ${ia.length} vs ${ib.length}`);
  let shown = 0;
  for (const id of ia) {
    for (const type of a.world.entities.componentTypes()) {
      const va = a.world.entities.get(id, type); const vb = b.world.entities.get(id, type);
      if (J(va) !== J(vb) && shown < 6) { shown += 1; out.push(`entity ${id} (${a.world.entities.kind(id)}) ${type.name}: ${J(va)?.slice(0, 300)} vs ${J(vb)?.slice(0, 300)}`); }
    }
  }
  if (J(a.world.fortress) !== J(b.world.fortress)) out.push(`fortress: ${J(a.world.fortress)} vs ${J(b.world.fortress)}`);
  if (J(a.world.rngState()) !== J(b.world.rngState())) out.push('rng streams differ');
  for (const sys of a.world.scheduler.inOrder()) {
    const sa = J(sys.serialize()); const sb = J(b.world.scheduler.byId(sys.id)?.serialize());
    if (sa !== sb) { let k = 0; while (k < sa.length && sa[k] === sb?.[k]) k += 1; out.push(`system ${sys.id} state differs at char ${k}: ...${sa.slice(Math.max(0, k - 120), k + 120)}... vs ...${sb?.slice(Math.max(0, k - 120), k + 120)}...`); }
  }
  return out.length > 0 ? out.join('\n      ') : 'no difference found in grids, entities, fortress, rng or system state';
}

interface Fork { at: number; why: string; w: TestWorld; state: FortressState; twin: TestWorld; twinState: FortressState; diverged: number | null; firstScriptAfter: number | null; checked: number; note: string }

describe('T-115 probe: long run', () => {
  it(`${SEED} peril ${PERIL} for ${DAYS} days`, () => {
    mkdirSync(OUT, { recursive: true });
    const name = `${SEED}-p${PERIL}-${TAG}`;
    const w = threeYearsWorld(SEED, PERIL);
    const state = setupThreeYearsFortress(w);
    const probe = new Probe(name);
    const hashes: string[] = [];
    const eventCounts: Record<string, number> = {};
    const rejected: Record<string, number> = {};
    let attacksInChunk = 0;
    let thrown: string | null = null;
    w.game.world.onEvents((events) => {
      for (const e of events) {
        eventCounts[e.type] = (eventCounts[e.type] ?? 0) + 1;
        if (e.type === 'AttackResolved') attacksInChunk += 1;
        if (e.type === 'CommandRejected') { const k = `${String((e['command'] as { type: string }).type)}:${String(e['reason'])}`; rejected[k] = (rejected[k] ?? 0) + 1; }
      }
    });
    const forks: Fork[] = [];
    const doneForks: Omit<Fork, 'w' | 'state' | 'twin' | 'twinState'>[] = [];
    let loadDiffShown = 0;
    const forkBudget: Record<string, number> = { random: 6, combat: 3, hostile: 3, alert: 3, caravan: 3, trade: 3, paused: 2 };
    let lcg = 12345;
    const rnd = (): number => { lcg = (lcg * 1103515245 + 12345) & 0x7fffffff; return lcg / 0x7fffffff; };
    const ent = w.game.world.entities;

    const maybeFork = (): void => {
      if (!SAVE || forks.length >= 1) return;
      let why: string | null = null;
      const hostile = ent.with(C.Hostile).some((id) => (ent.get(id, C.Creature)?.faction ?? '').startsWith('hostile:') && ent.get(id, C.Creature)?.cagedIn === undefined);
      const parties = ent.with(C.Party).map((id) => ent.get(id, C.Party)!);
      const caravan = parties.find((p) => p.purpose === 'caravan');
      if (attacksInChunk > 0 && (forkBudget['combat'] ?? 0) > 0) why = 'combat';
      else if (w.game.world.fortress.alert !== 'calm' && (forkBudget['alert'] ?? 0) > 0) why = 'alert';
      else if (hostile && (forkBudget['hostile'] ?? 0) > 0) why = 'hostile';
      else if (caravan && /trad/.test(caravan.state) && (forkBudget['trade'] ?? 0) > 0) why = 'trade';
      else if (caravan && (forkBudget['caravan'] ?? 0) > 0 && rnd() < 0.2) why = 'caravan';
      else if ((forkBudget['random'] ?? 0) > 0 && rnd() < 0.004) why = 'random';
      if (why === null) return;
      forkBudget[why] = (forkBudget[why] ?? 0) - 1;
      // The reference is a loaded twin, so that a load that changes the world once (T-131) does not hide everything else:
      // twin = load(main); fork = load(twin). twin and fork are then played side by side with the script on both.
      let twinGame: HeadlessGame; let loaded: HeadlessGame;
      try { twinGame = roundTrip(w.game).game; loaded = roundTrip(twinGame).game; } catch (e) { doneForks.push({ at: w.tick, why: `${why}: LOAD THREW ${(e as Error).message}`, diverged: w.tick, firstScriptAfter: null, checked: 0, note: '' }); return; }
      let note = '';
      if (twinGame.hash() !== w.game.hash()) {
        note = 'first load differs from the unsaved game';
        if (loadDiffShown < 3) { loadDiffShown += 1; note += `:\n      ${whyDiffer(w.game, twinGame)}`; }
      } else note = 'first load equals the unsaved game';
      const f: Fork = { at: w.tick, why, w: wrap(loaded), state: structuredClone(state), twin: wrap(twinGame), twinState: structuredClone(state), diverged: null, firstScriptAfter: null, checked: 0, note };
      if (loaded.hash() !== twinGame.hash()) { f.diverged = w.tick; f.note += `\n      second load differs from the first:\n      ${whyDiffer(twinGame, loaded)}`; }
      forks.push(f);
    };
    const stepForks = (n: number): void => {
      for (const f of forks) if (f.diverged === null) { f.w.game.step(n); f.twin.game.step(n); }
    };
    const compareForks = (): void => {
      for (let k = forks.length - 1; k >= 0; k -= 1) {
        const f = forks[k]!;
        if (f.diverged === null) {
          f.checked += 1;
          if (f.w.game.hash() !== f.twin.game.hash()) { f.diverged = w.tick; f.note += `\n      diverged in play:\n      ${whyDiffer(f.twin.game, f.w.game)}`; }
        }
        if (f.diverged !== null || w.tick - f.at >= FORK_TICKS) {
          doneForks.push({ at: f.at, why: f.why, diverged: f.diverged, firstScriptAfter: f.firstScriptAfter, checked: f.checked, note: f.note });
          forks.splice(k, 1);
        }
      }
    };

    const total = DAYS * TICKS_PER_DAY;
    const bootstrap = 20 * TICKS_PER_DAY;
    try {
      while (w.tick < total) {
        const chunk = w.tick < bootstrap ? Math.min(100, bootstrap - w.tick) : Math.min(300, total - w.tick);
        let left = chunk;
        attacksInChunk = 0;
        while (left > 0) {
          const n = Math.min(EVERY, left);
          stepThreeYears(w, n);
          stepForks(n);
          left -= n;
          probe.check(w.game);
          if (forks.length > 0) compareForks();
          if (w.tick % TICKS_PER_DAY === 0) hashes.push(`${w.tick} ${w.game.hash()}`);
        }
        script(w, state);
        for (const f of forks) if (f.diverged === null) { script(f.w, f.state); script(f.twin, f.twinState); if (f.firstScriptAfter === null) f.firstScriptAfter = w.tick; }
        maybeFork();
      }
    } catch (e) {
      thrown = `THROWN at tick ${w.tick}: ${(e as Error).stack ?? String(e)}`;
    }
    const dwarves = ent.with(C.Labors).filter((id) => ent.get(id, C.Creature)?.faction === 'fortress').length;
    const its = ent.with(C.Item).map((id) => ({ id, it: ent.get(id, C.Item)! }));
    const deadCarrier = its.filter((x) => x.it.carriedBy !== 0 && !ent.alive(x.it.carriedBy));
    const tally = `end state: items ${its.length}; carried by an entity that no longer exists ${deadCarrier.length} (of them foreign ${deadCarrier.filter((x) => (x.it.flags & 256) !== 0).length}, counted in stock ${deadCarrier.filter((x) => (x.it.flags & (256 | 64)) === 0).length}); with both a position and a carrier ${its.filter((x) => x.it.carriedBy !== 0 && ent.has(x.id, C.Position)).length}; owned by an entity that no longer exists ${its.filter((x) => x.it.ownerId !== 0 && !ent.alive(x.it.ownerId)).length}; party entities ${ent.with(C.Party).length} (state gone ${ent.with(C.Party).filter((id) => ent.get(id, C.Party)?.state === 'gone').length}); rooms owned by a dead dwarf ${ent.with(C.Room).filter((id) => { const o = ent.get(id, C.Room)!.ownerId; return o !== 0 && !ent.alive(o); }).length}`;
    const out = [
      `run ${name} ended at tick ${w.tick} (day ${Math.floor(w.tick / TICKS_PER_DAY)}), dwarves alive ${dwarves}, final hash ${w.game.hash()}`,
      thrown ?? 'no exception',
      tally,
      probe.report(),
      '-- forks (save/load): at, why, diverged tick, first script tick after the fork, comparisons',
      ...doneForks.map((f) => `fork at t${f.at} (${f.why}): ${f.diverged === null ? 'equal' : `DIVERGED by t${f.diverged}`}; first script after t${f.firstScriptAfter}; ${f.checked} comparisons; ${f.note}`),
      '-- rejected commands',
      JSON.stringify(rejected),
      '-- events',
      JSON.stringify(eventCounts),
    ].join('\n');
    writeFileSync(`${OUT}/${name}.txt`, out);
    writeFileSync(`${OUT}/${name}.hashes.txt`, hashes.join('\n'));
    console.log(out.split('\n').slice(0, 60).join('\n'));
  }, 3_600_000);
});
