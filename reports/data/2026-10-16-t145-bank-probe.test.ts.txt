/** T-145 diagnosis (not committed): can a dwarf stand beside the generated brook on its level? */
import { describe, it } from 'vitest';
import { FluidType, TileKind } from '@core/map3d';
import { threeYearsWorld } from '../scenarios/three_years_fortress';

describe('T-145 bank probe', () => {
  it('counts the stand tiles beside the brook', () => {
    const w = threeYearsWorld('three-years');
    const m = w.map;
    const kinds: Record<string, number> = {};
    let water = 0, withStand = 0, standAbove = 0;
    for (let t = 0; t < m.tileCount; t += 1) {
      if (m.fluidLevelAt(t) < 1 || m.fluidTypeAt(t) !== FluidType.Water || m.zOf(t) < 30) continue;
      water += 1;
      let stand = false, above = false;
      for (let dy = -1; dy <= 1; dy += 1) for (let dx = -1; dx <= 1; dx += 1) {
        if (!dx && !dy) continue;
        const s = m.offset(t, dx, dy, 0);
        if (s < 0) continue;
        const key = `z${m.zOf(t)}:${TileKind[m.kindAt(s)]}:fluid${m.fluidLevelAt(s)}`;
        kinds[key] = (kinds[key] ?? 0) + 1;
        if (w.readers.pathing.passable(s, 'walker')) stand = true;
        const up = m.offset(s, 0, 0, 1);
        if (up >= 0 && w.readers.pathing.passable(up, 'walker')) above = true;
      }
      if (stand) withStand += 1;
      if (above) standAbove += 1;
      const upHere = m.offset(t, 0, 0, 1);
      const key = `above-water:${TileKind[m.kindAt(upHere)]}`;
      kinds[key] = (kinds[key] ?? 0) + 1;
    }
    console.log(`[t145-bank] ${JSON.stringify({ water, withStandOnItsLevel: withStand, withWalkableBankOneLevelUp: standAbove, kinds, level: [...new Set(Array.from({ length: m.tileCount }, (_, t) => t).filter((t) => m.fluidLevelAt(t) > 0 && m.zOf(t) >= 30).map((t) => m.fluidLevelAt(t)))] })}`);
  });
});
