/** T-307: nobody is sent to a patient that has its attacker beside it. */
import { describe, expect, it } from 'vitest';
import { FLAT_TEST_SURFACE_Z as Z } from '@app/headless';
import { C } from '@core/components';
import type { JobsReader } from '@systems/jobs';
import { TestWorld } from '../../scenarios/test-world';

const NEEDS = ['hunger', 'thirst', 'rest', 'social', 'comfort', 'purpose', 'joy'] as const;

describe('T-307 a patient with its attacker beside it', () => {
  it('gets no medic until the animal is dead, and is bandaged then', () => {
    const w = new TestWorld({ seed: 't307-endangered-patient', world: 'flat-test', embark: 'none', surfaceWildlife: 'none' });
    const world = w.game.world;
    const jobs = w.readers.of<JobsReader>('11');
    const patient = w.spawnDwarf({ at: w.index(20, 20, Z) });
    const medic = w.spawnDwarf({ at: w.index(55, 20, Z), skills: { medicine: 20 } });
    for (const id of [patient, medic]) for (const need of NEEDS) w.setNeed(id, need, 0);
    w.spawnItem('cloth', 'flax_cloth', w.index(54, 20, Z), 5);
    w.step(2);
    const badger = w.spawnCreature('badger', w.index(21, 20, Z), 'wild');
    world.services.combat!.setTarget(badger, patient);
    world.emit({ type: 'DamageDealt', defenderId: patient, attackerId: badger, part: 'torso', damageType: 'edge', amount: 20 });
    w.until(() => world.entities.has(patient, C.Patient), 10);
    for (let tick = 0; tick < 30 && world.entities.alive(badger) && world.services.combat!.targetOf(badger) === patient; tick += 1) {
      w.step(1);
      // The job is on the board and nobody holds it: the medic stays where it is.
      expect(jobs.byKey(`stop_bleeding:${patient}`)?.workerId ?? 0).toBe(0);
    }
    expect(jobs.forWorker(medic)).toBeUndefined();
    world.emit({ type: 'DamageDealt', defenderId: badger, attackerId: patient, part: 'torso', damageType: 'edge',
      amount: world.entities.get(badger, C.Body)!.parts[1]!.maxHp * 2 });
    w.step(3);
    expect(world.entities.alive(badger)).toBe(false);
    w.until(() => jobs.byKey(`stop_bleeding:${patient}`)?.workerId === medic, 20);
    w.until(() => w.events('TreatmentDone').some((e) => e['patientId'] === patient && e['kind'] === 'stop_bleeding'), 900);
  });
});
