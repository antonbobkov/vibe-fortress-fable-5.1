/** Temporary probe (not committed): the T-176 findings of the peril 2 and 4 sweep seeds, without stopping. */
import { describe, it } from 'vitest';
import { appendFileSync } from 'node:fs';
const OUT = process.env['PROBE_OUT'] ?? '.vite/probe-sweep.txt';
const log = (line: string): void => { appendFileSync(OUT, line + String.fromCharCode(10)); };
import { TICKS_PER_DAY } from '@core/constants';
import { C } from '@core/components';
import { checkInvariants, watchInvariants } from '../scenarios/invariants';
import {
  playThreeYears, setupThreeYearsFortress, checkFortressWorkOrders, checkFortressRooms,
  checkFortressCaveFarm, checkFortressDefense, threeYearsWorld,
} from '../scenarios/three_years_fortress';
const THREE_YEARS_20_SEEDS = [1, 2, 3, 4, 5, 6, 7].map((n) => ({ seed: `three-years-p2-${n}`, peril: 2 }))
  .concat([1, 2, 3, 4, 5, 6].map((n) => ({ seed: `three-years-p4-${n}`, peril: 4 })));

const UNTIL_DAY = Number(process.env['PROBE_DAYS'] ?? '220');
const ONLY = process.env['PROBE_ONLY'];
const SEEDS = THREE_YEARS_20_SEEDS.filter((s) => s.peril > 0 && (ONLY === undefined || ONLY.split(',').includes(s.seed)));

describe('probe sweep', () => {
  it.each(SEEDS)('$seed', ({ seed, peril }) => {
    const w = threeYearsWorld(seed, peril);
    const seen = new Set<string>();
    const stop = watchInvariants(w.game, (findings) => {
      for (const f of findings) {
        const key = `${f.inv}:${f.cls}`;
        if (seen.has(key)) continue;
        seen.add(key);
        log(`[sweep] ${seed} day ${Math.floor(w.tick / TICKS_PER_DAY) + 1} ${JSON.stringify(f)}`);
      }
    });
    log(`[sweep-start] ${seed}`);
    const state = setupThreeYearsFortress(w);
    const boot = 20 * TICKS_PER_DAY;
    while (w.tick < boot) {
      playThreeYears(w, state, Math.min(100, boot - w.tick));
      checkFortressWorkOrders(w, state); checkFortressRooms(w, state); checkFortressCaveFarm(w, state); checkFortressDefense(w, state);
    }
    const end = UNTIL_DAY * TICKS_PER_DAY;
    while (w.tick < end) {
      playThreeYears(w, state, Math.min(300, end - w.tick));
      checkFortressWorkOrders(w, state); checkFortressRooms(w, state); checkFortressCaveFarm(w, state); checkFortressDefense(w, state);
    }
    stop();
    const living = w.game.world.entities.with(C.Labors).filter((id) => w.game.world.entities.get(id, C.Creature)?.faction === 'fortress').length;
    log(`[sweep-end] ${seed} day ${UNTIL_DAY} living ${living} findings ${checkInvariants(w.game).length} kinds ${[...seen].join(',') || 'none'}`);
  }, 3_600_000);
});
