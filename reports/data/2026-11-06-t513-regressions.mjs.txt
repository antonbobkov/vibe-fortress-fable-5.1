// usage: node inject.mjs <dir> <refShifts comma> <testShifts comma> <tolLiving> <tolDeaths>
// Judges synthetic regressions built from the test shifts against the reference shifts' pooled row.
import { readdirSync, readFileSync } from 'node:fs';
import { join } from 'node:path';
const [dir, refS, testS, tolL, tolD] = process.argv.slice(2);
const ref = refS.split(',').map(Number), test = testS.split(',').map(Number);
const load = (s) => readdirSync(join(dir, `shift-${s}`)).filter((f) => /^three-years-p\d-\d+\.json$/.test(f)).map((f) => JSON.parse(readFileSync(join(dir, `shift-${s}`, f), 'utf8')))
  .map(({ result, needDeaths }) => ({ seed: result.seed, peril: result.peril, y1: result.y1Pop, y3: result.y3Pop, deaths: result.dwarfDeaths, need: needDeaths ?? 0, ext: result.extinctDay ?? null }));
const fig = (rows) => { const p0 = rows.filter((r) => r.peril === 0); return { alive: rows.filter((r) => r.y1 > 0).length, grown: p0.filter((r) => r.y3 > 7).length, hungry: p0.filter((r) => r.need > 0).length, living: rows.reduce((t, r) => t + r.y3, 0), deaths: rows.reduce((t, r) => t + r.deaths, 0), extinct: rows.filter((r) => r.ext !== null).length }; };
const pool = (fs) => Object.fromEntries(Object.keys(fs[0]).map((k) => [k, Math.round(fs.reduce((t, f) => t + f[k], 0) / fs.length * 100) / 100]));
const refPool = pool(ref.map((s) => fig(load(s))));
const check = (p) => { const e = []; if (p.alive < 18) e.push('alive'); if (p.grown < 6) e.push('grown'); if (p.hungry > 1) e.push('hungry'); if (p.living < refPool.living - Number(tolL)) e.push('living'); if (p.deaths > refPool.deaths + Number(tolD)) e.push('deaths'); return e; };
console.log('reference', JSON.stringify(refPool), 'bounds living >=', refPool.living - Number(tolL), 'deaths <=', refPool.deaths + Number(tolD));
const base = test.map((s) => load(s));
const cases = [['none', () => {}]];
const starve = (seed, k, dead) => (rows, i) => { if (i < k) { const r = rows.find((x) => x.seed === seed); r.need = dead; r.deaths += dead; r.y3 = Math.max(0, r.y3 - dead); } };
const wipe = (seed, k, y1) => (rows, i) => { if (i < k) { const r = rows.find((x) => x.seed === seed); r.deaths += r.y3; r.y3 = 0; r.ext = 100; if (y1) r.y1 = 0; } };
const N = test.length;
for (const seed of ['three-years-p0-1', 'three-years-p0-5']) for (const k of [N, Math.ceil(N / 2), 1]) cases.push([`${seed} 23 starve in ${k} of ${N}`, starve(seed, k, 23)]);
cases.push([`p0-1 and p0-5 both starve (23) in ${N} of ${N}`, (rows, i) => { starve('three-years-p0-1', N, 23)(rows, i); starve('three-years-p0-5', N, 23)(rows, i); }]);
cases.push([`p0-1, p0-5 starve in ${Math.ceil(N / 2)}, p0-2 in ${Math.ceil(N / 2)}`, (rows, i) => { starve('three-years-p0-1', Math.ceil(N / 2), 5)(rows, i); starve('three-years-p0-5', Math.ceil(N / 2), 5)(rows, i); starve('three-years-p0-2', Math.ceil(N / 2), 5)(rows, i); }]);
for (const k of [N, Math.ceil(N / 2)]) cases.push([`p4-2 and p2-5 and p4-4 dead before year 1 in ${k} of ${N}`, (rows, i) => { wipe('three-years-p4-2', k, true)(rows, i); wipe('three-years-p2-5', k, true)(rows, i); wipe('three-years-p4-4', k, true)(rows, i); }]);
for (const k of [N, Math.ceil(N / 2)]) cases.push([`p0-4 stays at seven (no wave) in ${k} of ${N}`, (rows, i) => { if (i < k) { const r = rows.find((x) => x.seed === 'three-years-p0-4'); r.y3 = 7; } }]);
cases.push([`p0-4 and p0-6 stay at seven in ${N} of ${N}`, (rows) => { for (const s of ['three-years-p0-4', 'three-years-p0-6']) rows.find((x) => x.seed === s).y3 = 7; }]);
for (const d of [20, 40, 60, 80]) cases.push([`every sweep ${d} more deaths and ${d} fewer living`, (rows) => { let left = d; for (const r of rows.filter((x) => x.peril > 0)) { const t = Math.min(left, Math.max(0, r.y3 - 1), 6); r.y3 -= t; r.deaths += t; left -= t; } }]);
for (const d of [40, 80, 120]) cases.push([`every sweep ${d} fewer living (fewer immigrants), no deaths`, (rows) => { let left = d; for (const r of rows) { const t = Math.min(left, Math.max(0, r.y3 - 10), 10); r.y3 -= t; left -= t; } }]);
for (const [name, f] of cases) {
  const samples = base.map((rows, i) => { const c = rows.map((r) => ({ ...r })); f(c, i); return fig(c); });
  const p = pool(samples); const e = check(p);
  console.log(`| ${name} | ${p.alive}/${p.grown}/${p.hungry}/${p.living}/${p.deaths}/${p.extinct} | ${e.length ? 'blocked: ' + e.join(', ') : 'pass'} |`);
}
