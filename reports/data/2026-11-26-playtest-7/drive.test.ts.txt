/** Playtest 11-26 driver (not committed): the three-year script at peril 2 into Years 4 to 6, saving on fixed days and warnings. */
import { describe, it } from 'vitest';
import { mkdirSync, writeFileSync } from 'node:fs';
import { TICKS_PER_DAY, DAYS_PER_YEAR } from '@core/constants';
import { C } from '@core/components';
import { saveGame } from '@app/game-save';
import { playThreeYears, setupThreeYearsFortress, threeYearsWorld, checkFortressWorkOrders, checkFortressRooms,
  checkFortressCaveFarm, checkFortressDefense } from './three_years_fortress';

const SEED = process.env['PT_SEED'] ?? 'playtest-1126-p2';
const PERIL = Number(process.env['PT_PERIL'] ?? 2);
const OUT = process.env['PT_OUT'] ?? 'out/pt7';
const SAVE_DAYS = (process.env['PT_SAVES'] ?? '224,336,364,392,420,448,476,504,532,560,588,616,644,672').split(',').map(Number);
const END_DAY = Number(process.env['PT_END'] ?? 672);
const WARN_SAVES_FROM = Number(process.env['PT_WARN_FROM'] ?? 337);

describe('playtest 7 drive', () => {
  it('drives', async () => {
    mkdirSync(OUT, { recursive: true });
    const w = threeYearsWorld(SEED, PERIL);
    const state = setupThreeYearsFortress(w);
    const counts: Record<string, number> = {};
    const notable: string[] = [];
    const keep = /Raid|Siege|Threat|Immigr|Caravan|TradeCompleted|Died|Ambush|Thief|Cavern|Breach|Monster|DeepStirred|Deepsilver|Title|Whim|Artifact|Masterwork|Legendary|Sulk|Restless|WebSpun|FirstRunecaster|Summon|HallSpirit|Fell|Collapsed|Starvation|CaravanLost|Mishap|PatientAdmitted|Buried/;
    let saveFlag = '';
    w.game.world.onEvents((events) => {
      const day = Math.floor(w.tick / TICKS_PER_DAY) + 1;
      for (const e of events) {
        counts[e.type] = (counts[e.type] ?? 0) + 1;
        if (keep.test(e.type) && e.type !== 'ThreatNearby') {
          const { type, ...rest } = e as unknown as Record<string, unknown>;
          if (type === 'CreatureDied') {
            const r = rest as { cause?: string; species?: unknown; name?: string; faction?: string; killerName?: string };
            notable.push(`d${day} CreatureDied ${r.faction} ${String(r.species)} ${r.name ?? ''} cause=${r.cause} killer=${r.killerName ?? ''}`);
          } else notable.push(`d${day} ${String(type)} ${JSON.stringify(rest).slice(0, 200)}`);
        }
        if (e.type === 'ThreatWarning' || e.type === 'CavernBreached' || e.type === 'MonsterEmerged') saveFlag = `${e.type}`;
      }
    });
    const saves = new Set(SAVE_DAYS);
    const timing: string[] = [];
    let last = performance.now();
    const write = (): void => writeFileSync(`${OUT}/log.txt`, [`year length ${DAYS_PER_YEAR}`, ...timing, '', ...notable, '', JSON.stringify(counts, null, 1)].join('\n'));
    for (let day = 1; day <= END_DAY; day++) {
      const end = day * TICKS_PER_DAY;
      while (w.tick < end) {
        playThreeYears(w, state, Math.min(day <= 20 ? 100 : 300, end - w.tick));
        checkFortressWorkOrders(w, state);
        checkFortressRooms(w, state);
        checkFortressCaveFarm(w, state);
        checkFortressDefense(w, state);
        if (saveFlag !== '' && day >= WARN_SAVES_FROM) {
          const name = `d${day}-${saveFlag}`;
          writeFileSync(`${OUT}/${SEED}-${name}.bin`, await saveGame(w.game, { kind: 'manual', name: `pt7 ${name}` } as never));
          console.log(`saved ${name}`);
        }
        saveFlag = '';
      }
      const ent = w.game.world.entities;
      const pop = ent.with(C.Labors).filter((id) => ent.get(id, C.Creature)?.faction === 'fortress').length;
      const now = performance.now();
      if (day % 7 === 0) timing.push(`d${day} pop ${pop} ents ${ent.with(C.Position).length} items ${ent.with(C.Item).length} ms/day ${Math.round(now - last)}`);
      last = now;
      if (saves.has(day)) {
        writeFileSync(`${OUT}/${SEED}-d${day}.bin`, await saveGame(w.game, { kind: 'manual', name: `pt7 d${day}` } as never));
        console.log(`saved d${day} pop ${pop}`);
      }
      if (day % 7 === 0) write();
    }
    write();
  }, 14_400_000);
});
