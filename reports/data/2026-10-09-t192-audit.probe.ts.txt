import { describe,it,expect } from 'vitest';
import { writeFileSync } from 'node:fs';
import { C } from '@core/components';
import { MAX_CREATURES,TICKS_PER_DAY,DAYS_PER_SEASON,TICKS_PER_YEAR } from '@core/constants';
import { TileFlag,TileKind } from '@core/map3d';
import { FLAT_TEST_SPEC,buildFlatTestMap } from '@app/headless';
import { surfaceCap,farFromDwarves } from '@systems/animals';
import { TestWorld } from '../../test/scenarios/test-world';

describe('T-192 wild spawning measured against 08',()=>{it('runs both years, all seasons and contrasting biomes',()=>{
 const runs:unknown[]=[];
 for(const year of [1,2])for(const biome of ['temperate_grassland','taiga','glacier'])for(const season of [0,1,2,3]){
  const seed=`t192-audit-${biome}-${year}-${season}`;
  const w=new TestWorld({seed,map:buildFlatTestMap(undefined,64,64),embark:'none',spec:{...FLAT_TEST_SPEC,biome,traits:[]},startTick:(year-1)*TICKS_PER_YEAR+season*DAYS_PER_SEASON*TICKS_PER_DAY});
  const dwarf=w.spawnDwarf({at:w.index(8,32,32)});
  // Synthetic grass patch, not a year-long fortress fixture.
  for(let y=0;y<20;y++)for(let x=0;x<20;x++)w.map.setFlag(w.index(x,y,32),TileFlag.Grass,true);
  const eligible=w.animals.facts.surfaceSpecies(biome,season);
  const caps=Object.fromEntries(eligible.map(([s,weight])=>[w.animals.facts.idOf(s),surfaceCap(w.animals.spawner,s,weight)]));
  const spawns:unknown[]=[]; const violations:unknown[]=[];const days:unknown[]=[]; const totals:Record<string,number>={};let grazed=0;
  w.game.world.onEvents(batch=>{
   for(const ev of batch){if(ev.type==='AnimalSpawned'){
    const id=ev['id'] as number;const s=w.game.world.entities.get(id,C.Creature)?.species;const p=w.game.world.entities.get(id,C.Position);if(s===undefined||!p)continue;
    const tile=w.index(p.x,p.y,p.z);const species=w.animals.facts.idOf(s);const count=w.game.world.entities.with(C.Wild,C.Creature).filter(k=>w.game.world.entities.get(k,C.Creature)?.species===s).length;
    const group=spawns.filter((r:any)=>r.tick===w.tick&&r.species===species).length;
    const row={tick:w.tick,day:Math.floor(w.tick/TICKS_PER_DAY)+1,id,species,x:p.x,y:p.y,z:p.z,count,cap:caps[species],distanceOK:farFromDwarves(w.animals.spawner,tile),outdoor:w.map.hasFlag(tile,TileFlag.Outdoor),passable:w.pathing.reader().passable(tile,w.game.world.entities.get(id,C.Motion)!.movementClass),edge:p.x===0||p.y===0||p.x===63||p.y===63,groupMember:group};
    spawns.push(row);totals[species]=(totals[species]??0)+1;
    if(!eligible.some(([k])=>k===s)||count>caps[species]||!row.distanceOK||!row.outdoor||!row.passable||w.game.world.entities.with(C.Creature).length>MAX_CREATURES)violations.push(row);
   }if(ev.type==='GrassGrazed')grazed++;
   if(ev.type==='DayPassed')days.push({tick:w.tick,live:w.game.world.entities.with(C.Wild).length,counts:Object.fromEntries(eligible.map(([s])=>[w.animals.facts.idOf(s),w.game.world.entities.with(C.Wild,C.Creature).filter(k=>w.game.world.entities.get(k,C.Creature)?.species===s).length]))});
   }
  });
  w.step(10*TICKS_PER_DAY);
  runs.push({seed,year,biome,season,eligible:eligible.map(([s])=>w.animals.facts.idOf(s)),caps,totals,grazed,spawns,violations,days});
  console.log('T192 AUDIT',JSON.stringify({year,biome,season,totals,grazed,violations}));
 }
 writeFileSync('.vite/task-resume/t192-wild-audit.json',JSON.stringify(runs,null,2));
 expect(runs.length).toBe(24);
},300000);});

it('measures natural arrivals entering an open fortress room versus a sealed room',()=>{
 const results:unknown[]=[];
 for(const sealed of [false,true]){
  const w=new TestWorld({seed:'year2',map:buildFlatTestMap(undefined,64,64),embark:'none'});
  const interior:number[]=[];
  for(let y=4;y<=20;y++)for(let x=4;x<=20;x++){
   const t=w.index(x,y,32);const wall=x===4||x===20||y===4||y===20;
   if(wall && (sealed || x!==12 || y!==20))w.carve(t,TileKind.Wall);
   else if(!wall){interior.push(t);w.map.setFlag(t,TileFlag.Grass,true);}
  }
  w.zone('pasture',interior);w.build('bed',w.index(12,12,32),'oak');
  w.pathing.connectivity.rebuildAll();
  const entered=new Map<number,unknown>();const sightings:unknown[]=[];
  w.game.world.onEvents(batch=>{
   if(!batch.some(e=>e.type==='EntitiesMoved'))return;
   for(const id of w.game.world.entities.with(C.Wild,C.Position)){
    const p=w.game.world.entities.get(id,C.Position)!;const tile=w.index(p.x,p.y,p.z);
    if(interior.includes(tile)&&!entered.has(id)){
     const record={tick:w.tick,id,species:w.animals.facts.idOf(w.game.world.entities.get(id,C.Creature)!.species),position:p};entered.set(id,record);sightings.push(record);
    }
   }
  });
  w.step(30*TICKS_PER_DAY);
  results.push({sealed,spawnCount:w.events('AnimalSpawned').length,entered:sightings,grassRemaining:interior.filter(t=>w.map.hasFlag(t,TileFlag.Grass)).length});
 }
 writeFileSync('.vite/task-resume/t192-fortress-wandering.json',JSON.stringify(results,null,2));
 console.log('T192 FORTRESS',JSON.stringify(results));
});
