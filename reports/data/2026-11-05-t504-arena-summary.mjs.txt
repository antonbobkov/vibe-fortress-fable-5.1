import { readFileSync } from 'node:fs';
const rows = readFileSync(process.argv[2], 'utf8').split('\n').map((l) => l.trim()).filter((l) => l.startsWith('{')).map((l) => JSON.parse(l));
const by = new Map();
for (const r of rows) { if (!by.has(r.kind)) by.set(r.kind, []); by.get(r.kind).push(r); }
const avg = (a, f) => (a.reduce((s, r) => s + f(r), 0) / a.length).toFixed(2);
console.log('kind n | repelled wiped looted withdrew active | squadDown | soldDead shootDead civDead | raidDead | nerveFl bloodFl | captDown/died | escLoot | brokeLoss | friendlyHits | cascade(>=2 nerve before break)');
for (const [k, a] of by) {
  const c = (o) => a.filter((r) => r.outcome === o).length;
  console.log(`${k} ${a.length} | ${c('repelled')} ${c('wiped')} ${c('looted')} ${c('withdrew')} ${c('active')} | ${a.filter(r=>r.squadDown).length} | ${avg(a,r=>r.soldiersDead)} ${avg(a,r=>r.shootersDead)} ${avg(a,r=>r.civDead)} | ${avg(a,r=>r.raidersDead)} | ${avg(a,r=>r.nerveFlights)} ${avg(a,r=>r.bloodFlights)} | ${a.filter(r=>r.captainFate!=='standing').length} | ${avg(a,r=>r.escapedLoot)} | ${avg(a.filter(r=>r.brokeLosses>=0),r=>r.brokeLosses)} | ${avg(a,r=>r.raiderFriendlyHits)} | ${a.filter(r=>r.fledBeforeBreak>=2 && r.nerveFlights>=2).length}`);
}
