/**
 * T-512 probe (not kept in test/): one three-year seed played as the sweep plays it, counting by
 * year the surface crops planted, harvested and killed by frost, and the cave harvests.
 */
import { it } from 'vitest';
import { DAYS_PER_YEAR, TICKS_PER_DAY } from '@core/constants';
import { C } from '@core/components';
import type { PlantsReader } from '@systems/plants';
import type { BuildingsReader } from '@systems/buildings';
import { playThreeYears, setupThreeYearsFortress, checkFortressWorkOrders, checkFortressRooms, checkFortressCaveFarm, checkFortressDefense, threeYearsWorld } from '../scenarios/three_years_fortress';

const SEED = process.env['T512_SEED'] ?? 'three-years-p0-3';
const PERIL = Number(process.env['T512_PERIL'] ?? 0);
const SHIFT = Number(process.env['T512_SHIFT'] ?? 0);

it(`T-512 frost probe ${SEED} shift ${SHIFT}`, () => {
  const w = threeYearsWorld(SEED, PERIL);
  if (SHIFT > 0) w.step(SHIFT);
  const state = setupThreeYearsFortress(w);
  const ent = w.game.world.entities;
  const plants = w.readers.of<PlantsReader>('07');
  const rows: Record<string, number>[] = [{}, {}, {}];
  const add = (k: string, n = 1): void => {
    const y = Math.min(2, Math.floor(w.tick / TICKS_PER_DAY / DAYS_PER_YEAR));
    const r = rows[y]!;
    r[k] = (r[k] ?? 0) + n;
  };
  const surface = (tile: number): boolean => plants.plotKind(plants.plotAt(tile)) === 'surface';
  w.game.world.onEvents((events) => {
    for (const e of events) {
      if (e.type === 'CropHarvested') add(surface(Number(e['tile'])) ? 'surfHarvests' : 'caveHarvests');
      else if (e.type === 'CropPlanted' && surface(Number(e['tile']))) add('surfPlanted');
      else if (e.type === 'CropDied') add(`died:${String(e['reason'])}`);
      else if (e.type === 'CreatureDied' && ent.has(Number(e['id']), C.Labors)) add(`dwarfDied:${String(e['cause'])}`);
    }
  });
  const total = 3 * DAYS_PER_YEAR * TICKS_PER_DAY;
  const boot = 20 * TICKS_PER_DAY;
  while (w.tick < total) {
    playThreeYears(w, state, w.tick < boot ? Math.min(100, boot - w.tick) : Math.min(300, total - w.tick));
    checkFortressWorkOrders(w, state);
    checkFortressRooms(w, state);
    checkFortressCaveFarm(w, state);
    checkFortressDefense(w, state);
    if (w.tick % (14 * TICKS_PER_DAY) < 300) {
      const k = { caveTiles: 0, caveSown: 0, surfTiles: 0, surfSown: 0 };
      for (const z of plants.plots()) {
        const zone = w.readers.of<BuildingsReader>('14').zone(z);
        if (!zone) continue;
        const cave = plants.plotKind(z) === 'cave';
        for (const t of zone.tiles) {
          if (cave) k.caveTiles += 1; else k.surfTiles += 1;
          if (plants.cropAt(t) !== 0) { if (cave) k.caveSown += 1; else k.surfSown += 1; }
        }
      }
      const seed: Record<string, number> = {};
      const seedType = w.game.registry.table('items').index('seed');
      const mats = w.game.registry.table<{ id: string }>('materials');
      for (const id of ent.with(C.Item)) { const it = ent.get(id, C.Item)!; if (it.type === seedType) { const m = mats.byIndex(it.material)?.id ?? '?'; seed[m] = (seed[m] ?? 0) + it.count; } }
      const living = ent.with(C.Labors).filter((id) => ent.get(id, C.Creature)?.faction === 'fortress').length;
      console.log(`T512J day ${Math.floor(w.tick / TICKS_PER_DAY)} dwarves ${living} ${JSON.stringify(k)} seed ${JSON.stringify(seed)} plans ${JSON.stringify(state.economy.cavePlans.map((p) => p.tiles.length))}`);
    }
  }
  const living = ent.with(C.Labors).filter((id) => ent.get(id, C.Creature)?.faction === 'fortress').length;
  console.log(`T512 ${SEED} shift ${SHIFT} living ${living} ${JSON.stringify(rows)}`);
}, 3_600_000);
