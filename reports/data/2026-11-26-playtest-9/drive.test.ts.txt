/** Playtest 9 driver (not committed): the three-year script at peril 1 with a builder's own orders on top. */
import { describe, it } from 'vitest';
import { mkdirSync, writeFileSync } from 'node:fs';
import { TICKS_PER_DAY, DAYS_PER_YEAR, LOCAL_MAP_Z } from '@core/constants';
import { C } from '@core/components';
import { Map3D, TileKind, TileFlag } from '@core/map3d';
import { saveGame } from '@app/game-save';
import { HeadlessGame, regionSpec } from '@app/headless';
import { contentRegistry } from '@content/loader';
import { generateLocal } from '@systems/terrain/generate';
import { TestWorld } from './test-world';
import { attachThreeYearsMeasurements, observeImmigration, type ThreeYearsMeasurements } from './three_years_measurements';
import { playThreeYears, setupThreeYearsFortress, threeYearsEmbark, checkFortressWorkOrders, checkFortressRooms,
  checkFortressCaveFarm, checkFortressDefense, THREE_YEARS_RELIEF, type FortressState } from './three_years_fortress';

const SEED = process.env['PT_SEED'] ?? 'pt9-1';
const PERIL = Number(process.env['PT_PERIL'] ?? 1);
const WEALTH = Number(process.env['PT_WEALTH'] ?? 3);
const OUT = process.env['PT_OUT'] ?? 'out/pt9';
const SAVE_DAYS = (process.env['PT_SAVES'] ?? '56,112,168,224,280,336,392,448').split(',').map(Number);
const END_DAY = Number(process.env['PT_END'] ?? 448);
const BUILDER = process.env['PT_BUILDER'] !== '0';
const PRO = process.env['PT_PRO'] === '1';
let scriptErrors = 0;

function world(seed: string, peril: number): TestWorld {
  const registry = contentRegistry();
  const spec = regionSpec(seed, { peril, wealth: WEALTH as never, biome: 'temperate_forest', geology: 'mixed', traits: [], relief: THREE_YEARS_RELIEF,
    river: { class: 'brook', inflow: ['w'], outflow: 's' } } as never, registry);
  const map = new Map3D(48, 48, LOCAL_MAP_Z);
  const local = generateLocal(spec, map, registry);
  let instrument: ThreeYearsMeasurements | null = null;
  const systems = HeadlessGame.standardSystems(spec, local.layerStack, local).map((s) =>
    s.id === '21' ? observeImmigration(s, () => instrument) : s);
  const w = new TestWorld({ seed, registry, map, spec, systems, embark: threeYearsEmbark(registry, seed, peril) });
  instrument = attachThreeYearsMeasurements(w, seed);
  return w;
}

interface Builder { crafter: number; mason: number; engraver: number; engraver2?: number; quarry?: number[]; statuesPlaced: number; notes: string[] }

const failures: string[] = [];
function cmd(w: TestWorld, c: Record<string, unknown>): boolean {
  const r = w.game.world.applyCommand(c as never);
  if (!r.ok && c['type'] !== 'PlaceBuilding') failures.push(`${String(c['type'])}:${(r as { reason?: string }).reason}`);
  return r.ok;
}

function skillOf(w: TestWorld, id: number, skill: string): number {
  const idx = w.game.registry.table<{ id: string }>('skills').rows.findIndex((r) => r?.id === skill);
  return w.game.world.entities.get(id, C.Skills)?.level[idx] ?? 0;
}

function fortressDwarves(w: TestWorld): number[] {
  const ent = w.game.world.entities;
  return ent.with(C.Labors).filter((id) => ent.get(id, C.Creature)?.faction === 'fortress');
}

/** The builder's own orders, once a day. */
function builderDay(w: TestWorld, state: FortressState, b: Builder, day: number): void {
  const R = w.readers as unknown as { of<T>(id: string): T };
  const B = R.of<{ get(id: number): { state: string; footprint: number[] } | undefined; at(i: number): number; ghost(t: string, o: number, r: number): { reason: string | null }; bedroomOf(id: number): number; zone(id: number): { tiles: number[] } | undefined }>('14');
  const ent = w.game.world.entities;
  const dwarves = fortressDwarves(w).filter((id) => (ent.get(id, C.Creature)?.ageTicks ?? 0) >= 12 * DAYS_PER_YEAR * TICKS_PER_DAY);
  const best = (skill: string, not: number[]): number => {
    let pick = 0; let lv = -1;
    for (const id of dwarves) { if (not.includes(id)) continue; const l = skillOf(w, id, skill); if (l > lv) { lv = l; pick = id; } }
    return pick;
  };
  const name = (id: number): string => ent.get(id, C.Creature)?.name ?? String(id);
  // A dedicated stone crafter at the craftsdwarf's workshop with a long order behind the script's.
  const craft = state.placedIds['craftdwarf'];
  if ((b.crafter === 0 || ent.get(b.crafter, C.Creature) === undefined) && craft && B.get(craft)?.state === 'built') {
    const titled = dwarves.filter((id) => (ent.get(id, C.Title)?.title ?? 'none') !== 'none');
    b.crafter = best('crafting', []); void titled;
    cmd(w, { type: 'SetLabor', dwarfId: b.crafter, labor: 'crafts', on: true });
    cmd(w, { type: 'SetLabor', dwarfId: b.crafter, labor: 'hauling', on: false });
    cmd(w, { type: 'SetWorkshopWorker', workshopId: craft, dwarfId: b.crafter });
    cmd(w, { type: 'AddWorkOrder', workshopId: craft, order: { recipeId: 'craft_stone', mode: 'once', remaining: 999, priority: 5 } });
    b.notes.push(`d${day} crafter ${name(b.crafter)} crafting ${skillOf(w, b.crafter, 'crafting')}`);
  }
  // Statues at the mason: keep three in stock, place them in the halls.
  const mason = state.placedIds['mason'];
  if (b.mason === 0 && mason && B.get(mason)?.state === 'built' && day >= 20) {
    b.mason = 1;
    cmd(w, { type: 'AddWorkOrder', workshopId: mason, order: { recipeId: 'make_statue', mode: 'maintain', threshold: 2, priority: 5 } });
    b.notes.push(`d${day} statue order`);
  }
  const statuesInStock = ent.with(C.Item).filter((id) => { const it = ent.get(id, C.Item)!; return w.game.registry.table<{ id: string }>('items').rows[it.type]?.id === 'statue' && (it.flags & 128) === 0; }).length;
  if (day % 7 === 0 && day >= 40 && state.hearthHallSetup && state.diningHallSetup && statuesInStock > 0 && b.statuesPlaced < 8) {
    const halls = [...state.hearthHallTiles, ...state.diningHallTiles];
    for (const t of halls) {
      if (B.at(t) !== 0) continue;
      let walls = 0; let clear = true;
      for (let dy = -1; dy <= 1; dy++) for (let dx = -1; dx <= 1; dx++) { if (dx === 0 && dy === 0) continue; const n = w.map.offset(t, dx, dy, 0); if (n < 0) continue; if (B.at(n) !== 0) clear = false; const k = w.map.kindAt(n); if (k === TileKind.Wall) walls++; }
      if (!clear || walls < 5) continue;
      if (B.ghost('statue', t, 0).reason !== null) continue;
      if (cmd(w, { type: 'PlaceBuilding', buildingType: 'statue', origin: t, orientation: 0, materialChoice: 0, priority: 5 })) { b.statuesPlaced++; b.notes.push(`d${day} statue placed (${statuesInStock} in stock)`); break; }
    }
  }
  // PRO: a quarry west of the halls for the crafter's stone, from day 60.
  if (PRO && day === 60) {
    const mats = w.game.registry.table<{ id: string; class: string }>('materials');
    const q: number[] = [];
    for (let y = 12; y <= 45; y++) for (let x = 1; x <= 11; x++) { const t = w.index(x, y, state.hallZ); const c = mats.rows[w.map.materialAt(t)]?.class; if (w.map.kindAt(t) === TileKind.Wall && (c === 'stone' || c === 'ore' || c === 'gem')) q.push(t); }
    // a way in: the row y=12 from x=12 east to the first floor
    for (let x = 12; x <= 30; x++) { const t = w.index(x, 12, state.hallZ); if (w.map.kindAt(t) !== TileKind.Wall) break; q.push(t); }
    b.quarry = q;
    b.notes.push(`d${day} quarry ${q.length}: ${cmd(w, { type: 'Designate', kind: 'dig', priority: 5, tiles: q })}`);
  }
  // Engrave the halls and the Thane's bedroom once they are smoothed.
  if (day % 7 === 3 && day >= 60) {
    if (b.engraver === 0) {
      b.engraver = best('engraving', [b.crafter]);
      cmd(w, { type: 'SetLabor', dwarfId: b.engraver, labor: 'crafts', on: true });
      b.notes.push(`d${day} engraver ${name(b.engraver)}`);
    }
    const tiles = new Set<number>([...state.hearthHallTiles, ...state.diningHallTiles]);
    if (PRO) {
      if (!b.engraver2) { b.engraver2 = best('engraving', [b.crafter, b.engraver]); cmd(w, { type: 'SetLabor', dwarfId: b.engraver2, labor: 'crafts', on: true }); b.notes.push(`d${day} engraver2 ${name(b.engraver2)}`); }
      for (const rid of ent.with(C.Room)) { const z = ent.get(ent.get(rid, C.Room)!.zoneId, C.Zone); for (const t of z?.tiles ?? []) tiles.add(t); }
    }
    const thane = dwarves.find((id) => ent.get(id, C.Title)?.title === 'thane');
    if (thane) { const z = B.zone(B.bedroomOf(thane)); for (const t of z?.tiles ?? []) tiles.add(t); }
    const cand = new Set<number>();
    for (const t of tiles) for (let dy = -1; dy <= 1; dy++) for (let dx = -1; dx <= 1; dx++) { const n = w.map.offset(t, dx, dy, 0); if (n >= 0) cand.add(n); }
    const eng: number[] = [];
    for (const t of cand) if (w.map.hasFlag(t, TileFlag.Smoothed) && !w.map.hasFlag(t, TileFlag.Engraved) && (w.readers.of<{ designationAt(i: number): unknown }>('03').designationAt(t) === null) && B.at(t) === 0) eng.push(t);
    if (eng.length > 0 && cmd(w, { type: 'Designate', kind: 'engrave', priority: 5, tiles: eng })) b.notes.push(`d${day} engrave ${eng.length}`);
  }
}

describe('playtest 9 drive', () => {
  it('drives', async () => {
    mkdirSync(OUT, { recursive: true });
    const w = world(SEED, PERIL);
    const state = setupThreeYearsFortress(w);
    const b: Builder = { crafter: 0, mason: 0, engraver: 0, statuesPlaced: 0, notes: [] };
    console.log(`fortress wagon ${JSON.stringify(w.game.arrival?.wagon)}`);
    const counts: Record<string, number> = {};
    const notable: string[] = [];
    const skills = w.game.registry.table<{ id: string }>('skills');
    const keep = /Raid|Siege|Immigr|Caravan|Died|Ambush|Thief|Breach|Title|Masterwork|Legendary|Sulk|Whim|RoomQuality|HallSpirit|Milestone|ThoughtGained/;
    w.game.world.onEvents((events) => {
      const day = Math.floor(w.tick / TICKS_PER_DAY) + 1;
      for (const e of events) {
        counts[e.type] = (counts[e.type] ?? 0) + 1;
        const r = e as unknown as Record<string, unknown>;
        if (e.type === 'ItemCrafted' && Number(r['quality']) >= 4) { counts[`q${String(r['quality'])}`] = (counts[`q${String(r['quality'])}`] ?? 0) + 1; }
        if (e.type === 'MoodTierChanged') counts[`tier→${String(r['to'])}`] = (counts[`tier→${String(r['to'])}`] ?? 0) + 1;
        if (e.type === 'SkillLevelUp') {
          const id = Number(r['id']); const sk = skills.rows[Number(r['skill'])]?.id ?? String(r['skill']);
          if (id === b.crafter || id === b.engraver || Number(r['level']) >= 12) notable.push(`d${day} SkillLevelUp ${w.game.world.entities.get(id, C.Creature)?.name} ${sk} ${String(r['level'])}`);
          continue;
        }
        if (keep.test(e.type) && e.type !== 'ThreatNearby' && !(e.type === 'ThoughtGained' && /hearth_lit|raid_repelled/.test(String(r['thoughtId'])))) {
          const { type, ...rest } = r;
          notable.push(`d${day} ${String(type)} ${JSON.stringify(rest).slice(0, 200)}`);
        }
      }
    });
    const saves = new Set(SAVE_DAYS);
    const timing: string[] = [];
    let last = performance.now();
    const write = (): void => writeFileSync(`${OUT}/log.txt`, [`year length ${DAYS_PER_YEAR}`, ...timing, '', ...b.notes, '', `failures ${failures.slice(0, 20).join(' ')}`, '', ...notable, '', JSON.stringify(counts, null, 1)].join('\n'));
    for (let day = 1; day <= END_DAY; day++) {
      const end = day * TICKS_PER_DAY;
      while (w.tick < end) {
        playThreeYears(w, state, Math.min(day <= 20 ? 100 : 300, end - w.tick));
        for (const f of [checkFortressWorkOrders, checkFortressRooms, checkFortressCaveFarm, checkFortressDefense]) {
          try { f(w, state); } catch (err) { if (!PRO) throw err; scriptErrors++; if (scriptErrors < 20) b.notes.push(`d${day} script: ${String(err).slice(0, 200)}`); }
        }
      }
      if (BUILDER) builderDay(w, state, b, day);
      const pop = fortressDwarves(w).length;
      const now = performance.now();
      if (day % 7 === 0) {
        const extra = b.crafter ? ` crafter ${skillOf(w, b.crafter, 'crafting')} engraver ${b.engraver ? skillOf(w, b.engraver, 'engraving') : '-'}` : '';
        timing.push(`d${day} scriptErrors ${scriptErrors} pop ${pop} items ${w.game.world.entities.with(C.Item).length} ms/day ${Math.round(now - last)}${extra}`);
      }
      last = now;
      if (saves.has(day)) {
        writeFileSync(`${OUT}/${SEED}-d${day}.bin`, await saveGame(w.game, { kind: 'manual', name: `pt9 d${day}` } as never));
        console.log(`saved d${day} pop ${pop}`);
      }
      if (day % 7 === 0) write();
    }
    write();
  }, 14_400_000);
});
