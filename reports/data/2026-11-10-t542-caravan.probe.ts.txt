/**
 * T-542 probe (not for merge): what holds a caravan on its way to the depot. Every caravan that is
 * still walking to the depot a day after it arrived is dumped once a day: its depot, and each living
 * member's tile, destination, last walk outcome, labels, and a search with and without the mover.
 * Run with `T542_SEED=three-years-p2-2 THREE_YEARS_SHIFT=100`.
 */
import { it } from 'vitest';
import { TICKS_PER_DAY } from '@core/constants';
import { C } from '@core/components';
import type { BuildingsReader } from '@systems/buildings/reader';
import { TestWorld } from '../scenarios/test-world';
import { runThreeYearsSeed } from './three_years_seeds.test';

const SEED = process.env['T542_SEED'] ?? 'three-years-p2-2';
const PERIL = Number(/-p(\d)-/.exec(SEED)?.[1] ?? 2);

interface Active { partyId: number; civ: string; state: string; depotId: number; edgeTile: number; arrivedTick: number; stateSince: number;
  merchants: number[]; guards: number[]; animals: number[]; destinations: Record<number, number> }

it(`T-542 probe ${SEED}`, () => {
  const log: string[] = [];
  const original = TestWorld.prototype.step;
  const seen = new WeakSet<TestWorld>();
  const install = (w: TestWorld): void => {
    const map = w.game.world.map;
    const ent = w.game.world.entities;
    const xyz = (t: number): string => (t < 0 ? '-' : `${map.xOf(t)},${map.yOf(t)},${map.zOf(t)}`);
    const trade = w.game.world.scheduler.byId('20') as unknown as { active: Active | null };
    const movement = w.game.world.services.movement as unknown as {
      walks: Map<number, { target: number; request: number; following: boolean; attempts: number }>;
      moverOf(id: number): unknown; tileOf(id: number): number;
      pathing: { pathfinder: { searchImmediate(f: number, t: number, c: string, n?: number, m?: unknown): number[] | null } };
      walkTo(id: number, target: number, urgency: string, ctx: unknown): string;
    };
    const pr = w.readers.pathing;
    const buildings = w.readers.of<BuildingsReader>('14');
    const last = new Map<number, { status: string; tick: number; counts: Record<string, number> }>();
    const walkTo = movement.walkTo.bind(movement);
    movement.walkTo = (id, target, urgency, ctx) => {
      const out = walkTo(id, target, urgency, ctx);
      if (urgency === 'caravan') {
        const rec = last.get(id) ?? { status: '', tick: 0, counts: {} };
        rec.status = out;
        rec.tick = w.tick;
        rec.counts[out] = (rec.counts[out] ?? 0) + 1;
        last.set(id, rec);
      }
      return out;
    };
    let nextDump = 0;
    let dumps = 0;
    let lastParty = 0;
    const describeTile = (t: number): string => {
      if (t < 0) return '-';
      const b = buildings.at(t);
      const bv = b ? buildings.get(b) : undefined;
      return `kind ${map.kindAt(t)} pass ${pr.passable(t, 'walker') ? 'y' : 'n'} label ${pr.labelOf(t, 'walker')}${b ? ` bld ${b}:${bv?.type}:${bv?.state}` : ''}`;
    };
    const check = (): void => {
      const a = trade.active;
      if (a === null) return;
      if (a.partyId !== lastParty) {
        lastParty = a.partyId;
        nextDump = a.arrivedTick + TICKS_PER_DAY;
        log.push(`[t542] t${w.tick} day ${Math.floor(w.tick / TICKS_PER_DAY) + 1} caravan ${a.partyId} ${a.civ} arrived t${a.arrivedTick} state ${a.state} depot ${a.depotId} edge ${xyz(a.edgeTile)}`);
      }
      if (a.state !== 'toDepot' || w.tick < nextDump || dumps >= 40) return;
      nextDump = w.tick + (dumps < 6 ? TICKS_PER_DAY / 2 : 3 * TICKS_PER_DAY);
      dumps += 1;
      const depot = buildings.get(a.depotId);
      log.push(`[t542] DUMP t${w.tick} day ${Math.floor(w.tick / TICKS_PER_DAY) + 1} caravan ${a.partyId} ${a.civ} state ${a.state} since t${a.stateSince} alert ${w.game.world.fortress.alert} ` +
        `depot ${a.depotId} ${depot ? `${depot.type}:${depot.state} origin ${xyz(depot.origin)} footprint [${depot.footprint.map(xyz).join(' ')}]` : 'GONE'} depots now [${buildings.depots().join(',')}] edge ${xyz(a.edgeTile)} edgeLabel ${pr.labelOf(a.edgeTile, 'walker')}`);
      const roles: [string, number[]][] = [['merchant', a.merchants], ['guard', a.guards], ['animal', a.animals]];
      for (const [role, ids] of roles) for (const id of ids) {
        if (!ent.alive(id)) { log.push(`[t542]   ${role} ${id} dead`); continue; }
        const at = movement.tileOf(id);
        const dest = a.destinations[id] ?? -1;
        const rec = last.get(id);
        const walk = movement.walks.get(id);
        const pf = movement.pathing.pathfinder;
        const withMover = at < 0 || dest < 0 ? null : pf.searchImmediate(at, dest, 'walker', 200_000, movement.moverOf(id));
        const noMover = at < 0 || dest < 0 ? null : pf.searchImmediate(at, dest, 'walker', 200_000);
        const motion = ent.get(id, C.Motion) as unknown as { movementClass?: string; speed?: number } | undefined;
        const cr = ent.get(id, C.Creature) as unknown as { species?: number; faction?: string; cagedIn?: number } | undefined;
        log.push(`[t542]   ${role} ${id} sp ${cr?.species} fac ${cr?.faction} cls ${motion?.movementClass} speed ${motion?.speed} at ${xyz(at)} (${describeTile(at)}) dest ${xyz(dest)} (${describeTile(dest)}) ` +
          `last ${rec?.status ?? 'never'}@t${rec?.tick ?? -1} counts ${JSON.stringify(rec?.counts ?? {})} walk ${walk ? `${xyz(walk.target)} following ${walk.following} attempts ${walk.attempts}` : 'none'} ` +
          `reach ${at >= 0 && dest >= 0 ? pr.reachable(at, dest, 'walker') : '-'} search(mover) ${withMover === null ? 'none' : withMover.length} search ${noMover === null ? 'none' : noMover.length}` +
          `${withMover === null && noMover !== null ? ` blockedOnPath [${noMover.filter((t) => !pr.passable(t, 'walker', movement.moverOf(id) as never)).map((t) => `${xyz(t)}:${describeTile(t)}`).join(' ')}]` : ''}`);
      }
    };
    const step = w.step.bind(w);
    w.step = (n = 1): void => {
      for (let k = 0; k < n; k += 1) {
        step(1);
        if ((w.tick & 15) === 0) check();
      }
    };
  };
  TestWorld.prototype.step = function (this: TestWorld, n?: number) {
    if (!seen.has(this)) {
      seen.add(this);
      install(this);
      return this.step(n);
    }
    return original.call(this, n as number);
  } as typeof original;
  try {
    const r = runThreeYearsSeed(SEED, PERIL);
    console.log(`[t542] ${SEED} shift ${process.env['THREE_YEARS_SHIFT'] ?? '0'} y3Pop ${r.y3Pop} deaths ${r.dwarfDeaths}`);
  } finally {
    TestWorld.prototype.step = original;
    console.log(log.join('\n'));
  }
}, 3_600_000);
