/** Dispatch a burst of synthetic key events on the window in one task, as one frame's worth of input. */
async function burst(keys, label) {
  r.log(`burst of ${keys.length} synthetic keydown+keyup events dispatched on window in one task: ${label}`);
  const t0 = Date.now();
  await r.page.evaluate((list) => {
    for (const k of list) {
      const init = { key: k.key, bubbles: true, cancelable: true, shiftKey: !!k.shift, ctrlKey: !!k.ctrl };
      window.dispatchEvent(new KeyboardEvent('keydown', init));
      window.dispatchEvent(new KeyboardEvent('keyup', init));
    }
  }, keys);
  return Date.now() - t0;
}
/** Right-edge contact check on the sidebar and modal rows: rows whose last column holds text. */
async function edgeRows() {
  const st = await r.state();
  const rows = await r.text();
  const out = [];
  rows.forEach((x, i) => {
    const chars = [...x];
    const last = chars[chars.length - 1] ?? ' ';
    const prev = chars[chars.length - 2] ?? ' ';
    if (last !== ' ' && !'│█║┃'.includes(last) && prev !== ' ' && i < rows.length - 5) out.push(`${i}:${chars.slice(-12).join('').trim()}`);
  });
  return out;
}
