// Flow 5, part B: a trade the merchant refuses (caravan brought by the hook, as in flow 3 part C). Appends to flow 5.
import { launch, Recorder } from './common.mjs';
const s = await launch();
s.append = true;
const r = new Recorder('5-failure-paths', s);
await r.goto('/?start=flat-test&embark=default');
await r.snap('T0-start', 'A fresh default embark (hook URL ?start=flat-test&embark=default) for the trade refusal');
const fixture = await r.page.evaluate(() => {
  const h = window.__deepholm; const world = h.host.world; const registry = h.host.headless.registry;
  const granite = registry.table('materials').index('granite');
  const placed = world.applyCommand({ type: 'PlaceBuilding', buildingType: 'trade_depot', origin: h.readers.map.index(45, 8, 32), orientation: 0, materialChoice: 'granite' });
  if (!placed.ok) throw new Error(`depot: ${placed.reason}`);
  const depot = world.scheduler.byId('14').lastPlacedBuilding;
  world.scheduler.byId('14').completeNow(depot, granite, 0);
  const tile = h.readers.of('14').get(depot)?.footprint[0] ?? 0;
  world.applyCommand({ type: 'SetRequests', civ: 'hillfolk', entries: [{ itemType: 'cloth', priority: 3 }] });
  return { depot, tile };
});
r.log('hook setup: finished trade depot, request set');
const arrival = await r.page.evaluate(() => { const h = window.__deepholm; const trade = h.readers.of('20'); for (let n = 0; n < 120; n += 1) { h.step(5000); if (trade.activeCaravan()?.state === 'atDepot') return 'atDepot'; } return trade.activeCaravan()?.state ?? 'none'; });
r.log(`hook step until the caravan state is: ${arrival}`);
await r.page.evaluate(({ depot, tile }) => {
  const h = window.__deepholm; const registry = h.host.headless.registry;
  const type = registry.table('items').index('easy_meal'); const material = registry.table('materials').index('meal');
  const meals = [h.host.world.services.spawn.item(type, material, tile, 10, 1)];
  const hauler = h.dwarves()[0]; h.host.world.services.movement.teleport(hauler.id, tile);
  h.submit({ type: 'MoveToDepot', depotId: depot, itemIds: meals });
  const trade = h.readers.of('20');
  for (let n = 0; n < 30 && !meals.every((id) => trade.depotItems().includes(id)); n += 1) h.step(100);
  h.host.world.applyCommand({ type: 'AssignTitle', dwarfId: h.dwarves()[0].id, title: 'broker' });
}, fixture);
r.log('hook setup: meals in the depot, first dwarf made broker');
await r.key('Escape');
await r.key('c');
await r.key('Shift+D');
await r.key('Shift+T');
await r.page.evaluate(() => { const h = window.__deepholm; const trade = h.readers.of('20'); for (let n = 0; n < 300 && !trade.sessionOpen(); n += 1) h.step(1); });
r.log('hook step(1) until the trade session opens');
await r.snap('T1-session', 'Trade session open, nothing selected');
await r.key('Enter');
await r.snap('T2-offer-nothing', 'Key Enter with nothing selected on either side');
await r.key('ArrowDown');
await r.key('+');
await r.key('Enter');
await r.snap('T3-buy-offer-nothing', 'ArrowDown, +, Enter: take a merchant item and offer nothing for it');
await r.key('+');
await r.key('+');
await r.key('+');
await r.snap('T4-plus-plus-plus', 'Key + x3 on the merchant item');
await r.key('Enter');
await r.snap('T5-offer-more', 'Key Enter');
await r.key('g');
await r.snap('T6-gift', 'Key g (gift: offer with nothing taken)');
await r.key('Escape');
await r.snap('T7-escape', 'Key Escape');
await r.close();
