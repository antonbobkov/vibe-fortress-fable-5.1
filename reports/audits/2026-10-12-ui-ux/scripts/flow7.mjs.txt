// Flow 7: 70 days of a default embark, stepped, with no defensive preparation. Records every log-strip line and notice with its tick/day.
import { launch, Recorder, findSite, stepBy, closePanels } from './common.mjs';
import { generateWorldKeys, embarkAtRegion, siteFor, openingEconomy } from './economy.mjs';
import { writeFileSync } from 'node:fs';
import { join } from 'node:path';
const s = await launch();
const r = new Recorder('7-danger-guidance', s);
async function toFortress() { for (let i = 0; i < 6; i += 1) { const st = await r.state(); if (st.screen === 'fortress' && st.panel === null) break; await r.key('Escape'); } }

await generateWorldKeys(r, 'e2e-year');
await r.snap('A-world', 'World "e2e-year" generated by keyboard (the world the full-year spec plays)');
const target = await embarkAtRegion(r);
await r.snap('A-region', `Region chosen: ${JSON.stringify(target)}`);
await r.key('Enter');
await r.snap('A-embark-screen', 'Embark screen (default party and supplies; no preset change)');
await r.key('Enter');
await r.wait(3000);
await r.snap('A-start', 'Fortress opens paused at tick 400');
const site = await siteFor(r);
console.log('site', JSON.stringify({ ...site, trees: site.trees.length, patches: site.patches.length }));
await openingEconomy(r, site, 'E');

// ---- 70 days, stepped ----
const targetTick = 70 * 1200 + 400;
const seen = new Set();
const lines = [];   // every distinct log-strip / notice line with its first-seen tick
const events = [];  // selected events with tick
await r.page.evaluate(() => {
  const h = window.__deepholm;
  window.__t116Events = [];
  h.host.world.onEvents((evs) => { for (const e of evs) { const t = e.type; if (/Threat|Raid|Animal|Attack|Died|Door|Alert|Squad|Caravan|Immigr|Hostile|Siege|Wolf|Wild|Hunt|Flee|Starv|Hint/i.test(t)) window.__t116Events.push({ tick: h.readers.calendar.tick, type: t, species: e.species ?? e.speciesName ?? null, cause: e.cause ?? null, name: e.name ?? null, count: e.count ?? null }); } });
});
const dangerRe = /wolf|wolves|raid|alert|danger|attack|killed|dies|died|squad|door|weapon|armed|goblin|bandit|thief|thieves|wild|beast|predator|hostile|enem|flee|fled|hearth|watch|siege|animal|!!/i;
let snaps = 0;
const dayOf = (t) => Math.floor(t / 1200) + 1;
let guard = 0;
for (;;) {
  const out = await r.page.evaluate(({ target, seenArr }) => {
    const h = window.__deepholm;
    const seen = new Set(seenArr);
    const fresh = [];
    let tick = h.readers.calendar.tick;
    while (tick < target) {
      const slice = Math.min(100, target - tick);
      if (slice > 1) h.host.step(slice - 1);
      h.step(1);
      tick = h.readers.calendar.tick;
      const rows = h.text();
      const n = rows.length;
      const cand = [rows[n - 5], rows[n - 4], rows[n - 3], rows[n - 1]];
      for (const c of cand) { const t = c.replace(/\s+$/, ''); if (t.trim().length && !seen.has(t.trim())) { seen.add(t.trim()); if (!/^\s*\[k\] look/.test(t) && !/^\s*Day \d+ of /.test(t)) fresh.push({ tick, text: t.trim() }); } }
      if (fresh.length) break;
    }
    return { tick, fresh, seenArr: [...seen] };
  }, { target: targetTick, seenArr: [...seen] });
  for (const f of out.fresh) { lines.push({ tick: f.tick, day: dayOf(f.tick), text: f.text }); }
  for (const x of out.seenArr) seen.add(x);
  const notable = out.fresh.filter((f) => dangerRe.test(f.text) || f.text.startsWith('☼'));
  if (notable.length && snaps < 45) {
    snaps += 1;
    const slug = notable[0].text.replace(/[^A-Za-z0-9]+/g, '-').slice(0, 40).replace(/^-|-$/g, '');
    r.log(`hook step (60-tick slices) to tick ${out.tick}`);
    await r.snap(`D${String(dayOf(out.tick)).padStart(2, '0')}-${slug}`, `Day ${dayOf(out.tick)} (tick ${out.tick}): new line(s) on screen: ${notable.map((x) => JSON.stringify(x.text)).join(' ; ')}`, { settle: 60 });
  }
  guard += 1;
  if (out.tick >= targetTick || guard > 4000) break;
  // a pinned danger line: the first key after it only dismisses it, but we press nothing: the clock is stepped
}
const ev = await r.page.evaluate(() => window.__t116Events);
const pop = await r.page.evaluate(() => ({ dwarves: window.__deepholm.dwarves().length, tick: window.__deepholm.readers.calendar.tick, hash: window.__deepholm.hash() }));
console.log('population/tick', JSON.stringify(pop), 'lines', lines.length, 'events', ev.length);
writeFileSync(join(r.dir, 'log-lines-with-day.json'), JSON.stringify(lines, null, 1));
writeFileSync(join(r.dir, 'selected-events-with-day.json'), JSON.stringify(ev.map((e) => ({ ...e, day: dayOf(e.tick) })), null, 1));
await r.snap('Z-day70', `Tick ${pop.tick} (day ${dayOf(pop.tick)}): ${pop.dwarves} dwarves alive; world hash ${pop.hash}`);

// ---- what the game offers about danger, at day 70 ----
for (const [k, nm] of [['m', 'military'], ['t', 'threats'], ['n', 'animals'], ['a', 'log'], ['?', 'help'], ['c', 'civilizations']]) {
  await r.key(k);
  await r.snap(`Y-${nm}`, `Panel ${nm} (key ${k}) at tick ${pop.tick}`);
}
await r.key('?');
for (let i = 2; i <= 8; i += 1) { await r.key('PageDown'); await r.snap(`Y-help-page-${i}`, `Help panel PageDown (page ${i})`); }
// the Log panel in full
await toFortress();
await r.key('a');
await r.snap('X-log-1', 'Log panel, first page');
for (let i = 2; i <= 12; i += 1) { await r.key('PageDown'); await r.snap(`X-log-${i}`, `Log panel, PageDown (page ${i})`); }
await toFortress();
// ---- a child dwarf (immigrants at day 57 included a child): try to give it a labor (the failure path of flow 5) ----
const kid = await r.page.evaluate(() => { const d = window.__deepholm.readers.of('10'); const ids = d.list(); const idx = ids.findIndex((id) => d.isChild(id)); return { idx, n: ids.length, children: ids.filter((id) => d.isChild(id)).length }; });
r.log(`(setup read: dwarves ${kid.n}, children ${kid.children}, first child at list index ${kid.idx})`);
if (kid.idx >= 0) {
  await r.key('u');
  for (let i = 0; i < kid.idx; i += 1) await r.key('ArrowDown');
  await r.snap('W-child-row', `Units panel with the cursor on the first child (list index ${kid.idx})`);
  await r.key('l');
  await r.key('Enter');
  await r.snap('W-child-manager', 'Labors panel, Enter: labor manager (whose row is the cursor on?)');
  const kidName = await r.page.evaluate((i) => { const rows = window.__deepholm.text(); return rows[3 + i] ? [...rows[3 + i]].slice(46, 62).join('').trim() : ''; }, 0);
  void kidName;
  for (let i = 0; i < 14; i += 1) {
    const hdr = (await r.text())[1] ?? '';
    if (hdr.includes('Kari')) break;
    await r.key('ArrowDown');
  }
  await r.snap('W-child-manager-moved', 'ArrowDown in the labor manager until the header line names the child (Kari Thunderaxe, list index 8 of the units panel)');
  await r.key('Enter');
  await r.snap('W-child-toggle', 'Key Enter in the labor manager with the cursor on the child: toggle the first labor');
  await r.key('Space');
  await r.key('Enter');
  await r.snap('W-child-marked-toggle', 'Space (mark the child), Enter: toggle for the marked dwarf');
  await r.key('Escape');
  await toFortress();
}
await r.close();
