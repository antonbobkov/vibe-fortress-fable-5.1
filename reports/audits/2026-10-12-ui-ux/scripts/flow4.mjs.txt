import { launch, Recorder, waitText, findSite, stepBy, closePanels, embarkBitterbrae } from './common.mjs';
const s = await launch();
const r = new Recorder('4-newer-ui', s);
const T = async (name, fn) => { try { await fn(); } catch (e) { console.log(`!! ${name}: ${e.message}`); r.log(`(script error in ${name}: ${e.message})`); await r.snap(`error-${name}`, `Capture script error in ${name}: ${e.message}`); } };
async function toFortress() { for (let i = 0; i < 6; i += 1) { const st = await r.state(); if (st.screen === 'fortress' && st.panel === null) break; await r.key('Escape'); } }
const armBuild = async (cat, name) => { if ((await r.state()).panel !== 'build') await r.key('b'); await r.key(cat); await r.clickText(name); };
const ensureEditor = async (tile, needle) => { const rows = await r.text(); if (!rows.some((x) => x.includes(needle))) { if ((await r.state()).panel !== 'zones') await r.key('z'); await r.clickTile(tile.x, tile.y, 'zone tile'); } };
const downTo = async (needle, startRow) => { const rows = await r.text(); const idx = rows.findIndex((x, i) => i >= startRow && x.includes(needle)); return idx < 0 ? -1 : idx; };

// ---------- D. drag-to-place ----------
await T('drag', async () => {
  await r.key('b');
  await r.key('Shift+F');
  await r.snap('D-furniture', 'Key b, Shift+F: Furniture list');
  await r.clickText('Bed ');
  await r.snap('D-bed-tool', 'Clicked the "Bed" row');
  const row = { x: P[3].x, y: P[3].y };
  const a = await r.tileCell(row.x, row.y);
  const b = await r.tileCell(row.x + 4, row.y);
  const pa = await r.cellPx(a.col, a.row);
  const pb = await r.cellPx(b.col, b.row);
  r.log(`left-down at tile ${row.x},${row.y}; move to tile ${row.x + 4},${row.y} (not released)`);
  await r.page.mouse.move(pa.x, pa.y);
  await r.page.mouse.down();
  await r.page.mouse.move(pb.x, pb.y, { steps: 6 });
  await r.snap('D-bed-dragging', 'Mid-drag over five tiles with the Bed tool (button held): ghost footprints and the status row count');
  r.log('left-up');
  await r.page.mouse.up();
  await r.snap('D-bed-released', 'Released: beds placed, message about skipped tiles if any');
  await r.key('Escape');
  await toFortress();
  await armBuild('Shift+F', 'Door');

  const b2 = await r.tileCell(P[4].x + 4, P[4].y);

    const h = window.__deepholm; const sys = h.host.world.scheduler.byId('14');
    const b = h.readers.buildings;
    const out = [];
    for (const t of ['carpenter', 'bed']) for (const id of b.byType(t)) { const g = b.get(id); if (g && g.state !== 'built') { sys.completeNow(id, 1, 0); out.push(id); } }
