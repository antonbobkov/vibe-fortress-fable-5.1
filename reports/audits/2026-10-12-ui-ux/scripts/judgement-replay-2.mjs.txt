import { chromium } from '@playwright/test';
const BASE='http://127.0.0.1:5311';
const b = await chromium.launch({headless:true});
const p = await (await b.newContext({viewport:{width:1280,height:800}})).newPage();
await p.goto(BASE+'/?start=flat-test&embark=default'); await p.waitForFunction(()=>window.__deepholm&&window.__deepholm.screen()==='fortress');
await p.waitForTimeout(300);
const rows=async()=>p.evaluate(()=>window.__deepholm.text());
const cell=async(c,r)=>{const q=await p.evaluate(([a,b])=>window.__deepholm.cellCenter(a,b),[c,r]); const bx=await p.locator('canvas#screen').boundingBox(); return {x:bx.x+q.x,y:bx.y+q.y};};
const drag=async(a,bb)=>{const A=await cell(...a),B=await cell(...bb); await p.mouse.move(A.x,A.y); await p.mouse.down(); await p.mouse.move(B.x,B.y,{steps:6}); await p.mouse.up(); await p.waitForTimeout(100);};
const click=async(c,r,o)=>{const A=await cell(c,r); await p.mouse.click(A.x,A.y,o); await p.waitForTimeout(120);};
const key=async(k)=>{await p.keyboard.press(k); await p.waitForTimeout(80);};
const sel=async(l)=>{const r=await rows(); console.log(l, '=>', (r.find(x=>x.includes('Selected'))||'(no Selected line)').slice(44).trim(), '|', r[r.length-1].trim());};
await key('z'); await key('Shift+S'); await drag([10,10],[13,12]);
await key('Shift+F'); await drag([12,10],[16,12]);
await click(30,40,{button:'right'});
await p.evaluate(()=>window.__deepholm.step(2)); await p.waitForTimeout(100);
await click(10,10); await sel('tile 10,10 (stockpile only)');
await click(13,11); await sel('tile 13,11 (overlap)');
await click(16,11); await sel('tile 16,11 (farm only)');
await p.screenshot({path:'_t116replay/zones.png'});
await b.close();
